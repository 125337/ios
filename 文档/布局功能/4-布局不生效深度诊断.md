### 1.2 锤子助手 Hook 函数的精确流程

```objc
// FUN_00753950 的完整伪代码（逐行对应反编译）
- (id)hook_getValueOfProperty:(NSString *)property inRuleSet:(NSString *)ruleSet {
    // ── Stage 1: 检查是否开启了任一布局功能 ──
    BOOL globalOn = [WeChatTweakSettings enableChangeCSS];
    BOOL chatOn   = [WeChatTweakSettings enableChangeChatCSS];
    
    if (globalOn || chatOn) {
        // ── Stage 2: 全局布局匹配 ──
        BOOL matchedGlobal = NO;
        
        // 条件：globalOn + (allLevel 或 webLevel) + _font_set
        if (globalOn) {
            BOOL isAllLevel  = [property isEqualToString:@"allLevel"];
            BOOL isWebLevel  = [property isEqualToString:@"webLevel"];
            BOOL isFontSet   = [ruleSet hasSuffix:@"_font_set"];
            
            if ((isAllLevel || isWebLevel) && isFontSet) {
                matchedGlobal = YES;
            }
        }
        
        if (matchedGlobal) {
            // ── Stage 3: 获取CSS字符串并验证 ──
            NSString *cssStr = [WeChatTweakSettings changedCSSStr];
            
            if ([cssStr length] > 0) {
                // 从CSS中提取font size值（JSON解析或直接取值）
                // 验证 >= 10
                // 验证 <= 16
                
                // ── Stage 4: 调用原方法获取原始返回值 ──
                id originalResult = [self orig_getValueOfProperty:property inRuleSet:ruleSet];
                
                // ── Stage 5: 使用CSS值修改原始结果 ──
                id mutableCopy = [originalResult mutableCopy];
                // 用CSS值替换mutableCopy中的对应值
                // ... 修改操作 ...
                
                return mutableCopy;
            }
        }
        
        // ── Stage 6: 对话布局匹配 ──
        // 同上，但用 chatLevel + _font_set + enableChangeChatCSS
        // 用 changedChatCSSStr
    }
    
    // ── Stage 7: 都不匹配 → 走原方法 ──
    return [self orig_getValueOfProperty:property inRuleSet:ruleSet];
}
```

## 二、当前实现与锤子助手的逐行对比

### 2.1 CLocalInfo Hook 对比

| 环节 | 锤子助手 | 本项目 | 是否一致 |
|------|---------|--------|---------|
| 读取设置 | `[WeChatTweakSettings enableChangeCSS]`（直接从 NSUserDefaults 读） | `[FontLayoutConfig shared].globalLayoutEnabled`（从 ConfigManager / NSUserDefaults 读） | 逻辑等价 |
| 判断逻辑 | 任一开启 → return 1 | 任一开启 → return 1 | ✅ |
| 返回值 | `1` (unsigned int) | `1` (unsigned int) | ✅ |
| 回退 | 走原方法 | 走原方法 | ✅ |

### 2.2 MMThemeManager Hook 对比

| 环节 | 锤子助手 | 本项目 | 是否一致 |
|------|---------|--------|---------|
| 匹配属性 | `allLevel` / `webLevel` / `chatLevel` | `allLevel` / `webLevel` / `chatLevel` | ✅ |
| 匹配规则集 | `_font_set`（hasSuffix） | `_font_set`（hasSuffix） | ✅ |
| 读取配置 | `[WeChatTweakSettings changedCSSStr]`（从 NSUserDefaults 读 JSON） | `[FontLayoutConfig shared].globalFontSize`（CGFloat 值） | 方式不同 |
| 获取原始值 | **先调原方法** `orig_getValueOfProperty:inRuleSet:` | **不调原方法** | ❌ **关键差异** |
| 返回方式 | 对原结果 mutableCopy + 修改后返回 | 直接 return `@"%.0f"` | ❌ **关键差异** |

### 2.3 两个关键差异的影响

#### 差异1：不调原方法

```objc
// 锤子助手
id originalResult = [self orig_getValueOfProperty:property inRuleSet:ruleSet];
// 在 originalResult 基础上修改

// 本项目
return [NSString stringWithFormat:@"%.0f", fontSize];
// 直接返回新值，跳过原方法
```

**潜在问题**：如果 `getValueOfProperty:inRuleSet:` 有副作用，或者返回的对象不是简单 NSString，直接返回新字符串可能导致主题系统无法正确解析。

#### 差异2：存储方式不同

```objc
// 锤子助手
// 用户在UI中输入 JSON 字符串（如 {"fontSize":16}）
// 存储为 changedCSSStr → 读取后解析取值
NSString *cssStr = [WeChatTweakSettings changedCSSStr];
// 解析 cssStr 获取 font_size 值

// 本项目
// 用户在UI中输入数值（如 14）
// 存储为 globalFontSize (CGFloat) → ConfigManager 读写
CGFloat fontSize = config.globalFontSize;
// 直接使用数值
```

**潜在问题**：值是否正确写入并读取？

## 四、可能的问题根源

### 4.1 问题A：Hook 安装时机不对

`MioModuleRegistry registerAll` 可能在 WeChat 初始化完主题系统后才被调用，此时 `MMThemeManager` 已经加载了所有主题值，后续 Hook 不再生效。

**检查方式**：看日志中 `[MMThemeManager] CALLED` 是否出现。如果没有出现，说明 Hook 安装后 MMThemeManager 不再被调用。

**解决方案**：如果出现此问题，需要确保 Hook 在 MMThemeManager 首次使用之前安装。

### 4.2 问题B：return 值类型不对

`getValueOfProperty:inRuleSet:` 的返回值可能不是 NSString 而是 NSNumber 或其他类型。直接返回 `[NSString stringWithFormat:]` 可能导致类型不匹配。

**测试**：在 Hook 函数中记录原方法返回值的类型：

```objc
// 在 fallback 路径中记录原方法返回值的类型
id result = orig_getValueOfProperty_inRuleSet(self, _cmd, property, ruleSet);
WPLog(@"FontLayout", @"[ORIG] property=%@ ruleSet=%@ -> type=%@ value=%@",
      property, ruleSet, [result class], result);
return result;
```

### 4.3 问题C：m_uiGlobalFontLevel 被缓存

如果 WeChat 只在启动时读取一次 `m_uiGlobalFontLevel` 并缓存，后续不再调用此方法。我们的 Hook 返回 1 但可能为时已晚。

**检查方式**：在 `hook_m_uiGlobalFontLevel` 中加日志：

```objc
static unsigned int hook_m_uiGlobalFontLevel(id self, SEL _cmd) {
    FontLayoutConfig *config = [FontLayoutConfig shared];
    unsigned int result;
    
    if (config.globalLayoutEnabled || config.chatLayoutEnabled) {
        result = 1;
        WPLog(@"FontLayout", @"[CLocalInfo] m_uiGlobalFontLevel -> 1 (大字模式)");
    } else {
        result = orig_m_uiGlobalFontLevel(self, _cmd);
        WPLog(@"FontLayout", @"[CLocalInfo] m_uiGlobalFontLevel -> %d (原方法)", result);
    }
    
    return result;
}
```

### 4.4 问题D：ConfigManager 数据不同步

`FontLayoutConfig` 从 `ConfigManager` 读取的值可能与用户设置的值不一致（module prefix 拼接问题）。

**验证方式**：在 Hook 安装时直接打印 UserDefaults 原始值：

```objc
// 直接读取 NSUserDefaults 验证
NSUserDefaults *ud = [NSUserDefaults standardUserDefaults];
BOOL globalEnabled = [ud boolForKey:@"FontLayout_globalLayoutEnabled"];
CGFloat globalSize = [ud floatForKey:@"FontLayout_globalLayoutSize"];
WPLog(@"FontLayout", @"[UD Direct] globalEnabled=%d globalSize=%.0f",
      globalEnabled, globalSize);
```

## 五、最可能的故障原因（按概率排序）

| 概率 | 原因 | 现象 | 确认方法 |
|------|------|------|---------|
| ⭐⭐⭐⭐⭐ | **Hook 根本未触发**（Hook 安装失败或 `install` 未被调用） | 日志中无任何 `[MMThemeManager] CALLED` 输出 | 看日志 |
| ⭐⭐⭐⭐ | **MMThemeManager 类/方法不存在** | install 日志显示 `NIL` 或 `SKIPPED` | 加 install 日志 |
| ⭐⭐⭐ | **m_uiGlobalFontLevel 被 WeChat 缓存** | `[MMThemeManager] CALLED` 出现但界面无变化 | 看 m_uiGlobalFontLevel 日志 |
| ⭐⭐ | **返回值类型不匹配** | Hook 触发、日志正确、但界面无变化 | 记录原方法返回值类型 |
| ⭐ | **ConfigManager 值与配置不一致** | 日志显示配置值与预期不符 | 直接读 UserDefaults |

## 六、最简修复方案（同时解决所有可能问题）

如果你确定了是 "Hook 未触发" 或 "返回值类型" 问题，可以尝试以下修正后的 Hook.m：

```objc
// FontLayoutHook.m —— 修复版
// 关键改动：1. 调用原方法  2. 返回值类型与原方法一致

#import "FontLayoutHook.h"
#import "FontLayoutConfig.h"
#import "../../Core/LogManager.h"
#import <substrate.h>
#import <objc/runtime.h>

#pragma mark - 映射函数

/// 将布局大小（CGFloat 10-16）转换为原方法返回值的相同类型
/// 先调原方法获取类型，再创建同类型但带新值的对象
static id createValueWithFontSize(id originalValue, CGFloat fontSize) {
    if (originalValue == nil) {
        return [NSString stringWithFormat:@"%.0f", fontSize];
    }
    
    NSString *className = NSStringFromClass([originalValue class]);
    NSString *fontSizeStr = [NSString stringWithFormat:@"%.0f", fontSize];
    
    // 与原方法返回相同类型
    if ([className containsString:@"NSString"] || [className containsString:@"String"]) {
        // 原方法返回字符串 → 返回字符串
        return fontSizeStr;
    } else if ([className containsString:@"NSNumber"] || [className containsString:@"Number"]) {
        // 原方法返回数字 → 返回数字
        return @((NSInteger)fontSize);
    }
    
    // 默认返回字符串
    return fontSizeStr;
}

#pragma mark - Hook 1: MMThemeManager

static id (*orig_getValueOfProperty_inRuleSet)(id, SEL, NSString *, NSString *);

static id hook_getValueOfProperty_inRuleSet(id self, SEL _cmd,
                                             NSString *property,
                                             NSString *ruleSet) {
    FontLayoutConfig *config = [FontLayoutConfig shared];
    BOOL globalOn = config.globalLayoutEnabled;
    BOOL chatOn   = config.chatLayoutEnabled;

    // 先调原方法获取原始值（关键：与锤子助手一致）
    id originalResult = orig_getValueOfProperty_inRuleSet(self, _cmd, property, ruleSet);

    if (globalOn || chatOn) {
        // ── 全局布局 ──
        if (globalOn && [ruleSet hasSuffix:@"_font_set"]) {
            if ([property isEqualToString:@"allLevel"] ||
                [property isEqualToString:@"webLevel"]) {
                
                CGFloat fontSize = config.globalFontSize;
                if (fontSize >= 10.0f && fontSize <= 16.0f) {
                    id newValue = createValueWithFontSize(originalResult, fontSize);
                    return newValue;
                }
            }
        }

        // ── 对话布局 ──
        if (chatOn && [ruleSet hasSuffix:@"_font_set"]) {
            if ([property isEqualToString:@"chatLevel"]) {
                CGFloat fontSize = config.chatFontSize;
                if (fontSize >= 10.0f && fontSize <= 16.0f) {
                    id newValue = createValueWithFontSize(originalResult, fontSize);
                    return newValue;
                }
            }
        }
    }

    return originalResult;
}

#pragma mark - Hook 2: CLocalInfo

static unsigned int (*orig_m_uiGlobalFontLevel)(id, SEL);

static unsigned int hook_m_uiGlobalFontLevel(id self, SEL _cmd) {
    FontLayoutConfig *config = [FontLayoutConfig shared];
    if (config.globalLayoutEnabled || config.chatLayoutEnabled) {
        return 1;
    }
    return orig_m_uiGlobalFontLevel(self, _cmd);
}

#pragma mark - 安装

@implementation FontLayoutHook

+ (void)install {
    // ── 诊断日志 ──
    Class c1 = objc_getClass("MMThemeManager");
    Class c2 = objc_getClass("CLocalInfo");
    
    SEL s1 = @selector(getValueOfProperty:inRuleSet:);
    SEL s2 = @selector(m_uiGlobalFontLevel);
    
    WPLog(@"FontLayout", @"=== LayoutHook Install ===");
    WPLog(@"FontLayout", @"MMThemeManager: %@", c1 ? @"EXISTS" : @"NIL");
    WPLog(@"FontLayout", @"CLocalInfo: %@", c2 ? @"EXISTS" : @"NIL");
    WPLog(@"FontLayout", @"getValueOfProperty:inRuleSet: %@",
          class_getInstanceMethod(c1, s1) ? @"EXISTS" : @"NIL");
    WPLog(@"FontLayout", @"m_uiGlobalFontLevel: %@",
          class_getInstanceMethod(c2, s2) ? @"EXISTS" : @"NIL");
    
    FontLayoutConfig *config = [FontLayoutConfig shared];
    WPLog(@"FontLayout", @"Config: globalOn=%d globalSize=%.0f chatOn=%d chatSize=%.0f",
          config.globalLayoutEnabled, config.globalFontSize,
          config.chatLayoutEnabled, config.chatFontSize);
    
    // ── Hook ──
    if (c1 && class_getInstanceMethod(c1, s1)) {
        MSHookMessageEx(c1, s1,
                        (IMP)hook_getValueOfProperty_inRuleSet,
                        (IMP *)&orig_getValueOfProperty_inRuleSet);
        WPLog(@"FontLayout", @"[+] MMThemeManager hooked");
    } else {
        WPLog(@"FontLayout", @"[-] MMThemeManager SKIPPED");
    }
    
    if (c2 && class_getInstanceMethod(c2, s2)) {
        MSHookMessageEx(c2, s2,
                        (IMP)hook_m_uiGlobalFontLevel,
                        (IMP *)&orig_m_uiGlobalFontLevel);
        WPLog(@"FontLayout", @"[+] CLocalInfo hooked");
    } else {
        WPLog(@"FontLayout", @"[-] CLocalInfo SKIPPED");
    }
    
    WPLog(@"FontLayout", @"=== LayoutHook Install Complete ===");
}

@end
```
