## 方法源码

```objc
// L818-L834
+ (BOOL)wp_isCurrentDarkMode {
    if (@available(iOS 13.0, *)) {
        UIWindow *window = nil;
        if (@available(iOS 15.0, *)) {
            NSArray<UIWindow *> *windows = [UIApplication sharedApplication].connectedScenes
                .allObjects.firstObject.windows;
            window = windows.firstObject;
        } else {
            window = [UIApplication sharedApplication].keyWindow;
        }
        return window.traitCollection.userInterfaceStyle == UIUserInterfaceStyleDark;
    }
    return NO;
}
```

**被替代的原因**：当前代码中暗黑模式判断统一使用 `PluginConfig` 实例方法：

```objc
// wp_applyBorderAndBg:
UIColor *borderColor = isDark ? [UIColor colorWithHex:config.listCellBorderColorDarkHex]
                              : [UIColor colorWithHex:config.listCellBorderColor];

// 调用者已通过 [config isDarkMode] 或 [config isDarkModeForViewController:vc] 获取暗色状态
// 并作为 isDark 参数传入，无需 wp_isCurrentDarkMode
```
