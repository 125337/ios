### 1.3 创建实现文件

**新建文件**: `WeChatPlugin/Modules/HideAvatar/HideAvatarHook.m`

```objc
//
//  HideAvatarHook.m — 隐藏头像
//  复刻微信优化 1.6.5 isShowHeadImage Hook (FUN_000078bc)
//  架构: Hook CommonMessageViewModel.isShowHeadImage
//        Hook BaseMsgContentViewController.viewWillAppear: 用于聊天类型分类
//  MRC 编译
//

#import "HideAvatarHook.h"
#import "../../Config/PluginConfig.h"
#import "../../Core/LogManager.h"
#import <substrate.h>
#import <objc/runtime.h>
#import <objc/message.h>

// ============================================================
// MARK: - 聊天类型常量
// ============================================================
typedef NS_ENUM(NSInteger, MOChatType) {
    MOChatTypeUnknown       = 0,
    MOChatTypePrivate       = 1,  // 私聊
    MOChatTypeGroup         = 2,  // 群聊
    MOChatTypeOfficial      = 3,  // 公众号
};

// ============================================================
// MARK: - 全局状态
// ============================================================
static MOChatType _currentChatType = MOChatTypeUnknown;
static NSString  *_cachedUsername  = nil;  // 用于缓存去重

// ============================================================
// MARK: - Original IMPs
// ============================================================
static IMP _orig_BaseMsgContentVC_viewWillAppear     = NULL;
static IMP _orig_CommonMessageVM_isShowHeadImage     = NULL;

// ============================================================
// MARK: - Hook: BaseMsgContentViewController.viewWillAppear:
// ============================================================

static void hook_viewWillAppear(id self, SEL _cmd, BOOL animated) {
    // 调用原始实现
    ((void (*)(id, SEL, BOOL))_orig_BaseMsgContentVC_viewWillAppear)(self, _cmd, animated);

    // 获取联系人 username 并分类聊天类型
    id contact = nil;
    @try {
        // 通过 KVC 获取 m_contact
        contact = [self valueForKey:@"m_contact"];
    } @catch (NSException *e) {
        _currentChatType = MOChatTypeUnknown;
        return;
    }

    if (!contact) {
        _currentChatType = MOChatTypeUnknown;
        return;
    }

    NSString *username = nil;
    @try {
        username = [contact valueForKey:@"m_nsUsrName"];
    } @catch (NSException *e) {
        _currentChatType = MOChatTypeUnknown;
        return;
    }

    if (!username || username.length == 0) {
        [_cachedUsername release];
        _cachedUsername = nil;
        _currentChatType = MOChatTypeUnknown;
        return;
    }

    // 缓存去重: 同一个聊天页重复进入时不重新判定
    if (_cachedUsername && [username isEqualToString:_cachedUsername]) {
        return;
    }

    // 更新缓存
    [_cachedUsername release];
    _cachedUsername = [username copy];

    // 分类判定 (优先级: gh_ > @chatroom > 默认私聊)
    if ([username hasPrefix:@"gh_"]) {
        _currentChatType = MOChatTypeOfficial;
    } else if ([username containsString:@"@chatroom"]) {
        _currentChatType = MOChatTypeGroup;
    } else {
        _currentChatType = MOChatTypePrivate;
    }
}

// ============================================================
// MARK: - Hook: CommonMessageViewModel.isShowHeadImage
// ============================================================

static BOOL hook_isShowHeadImage(id self, SEL _cmd) {
    PluginConfig *config = [PluginConfig shared];

    // ── 快速路径: 6 个配置全关, 直接走原始 ──
    if (!config.hideC2COtherAvatar   &&
        !config.hideC2CSelfAvatar    &&
        !config.hideGroupOtherAvatar &&
        !config.hideGroupSelfAvatar  &&
        !config.hideOAOtherAvatar    &&
        !config.hideOASelfAvatar) {
        return ((BOOL (*)(id, SEL))_orig_CommonMessageVM_isShowHeadImage)(self, _cmd);
    }

    // ── 聊天类型未知, 不干预 ──
    if (_currentChatType == MOChatTypeUnknown) {
        return ((BOOL (*)(id, SEL))_orig_CommonMessageVM_isShowHeadImage)(self, _cmd);
    }

    // ── 获取 isSender ──
    BOOL isSender = NO;
    @try {
        isSender = [[self valueForKey:@"isSender"] boolValue];
    } @catch (NSException *e) {
        return ((BOOL (*)(id, SEL))_orig_CommonMessageVM_isShowHeadImage)(self, _cmd);
    }

    // ── 查表: (聊天类型, isSender) → 对应配置项 ──
    BOOL shouldHide = NO;

    switch (_currentChatType) {
        case MOChatTypePrivate:  // 私聊
            shouldHide = isSender ? config.hideC2CSelfAvatar
                                  : config.hideC2COtherAvatar;
            break;

        case MOChatTypeGroup:    // 群聊
            shouldHide = isSender ? config.hideGroupSelfAvatar
                                  : config.hideGroupOtherAvatar;
            break;

        case MOChatTypeOfficial: // 公众号
            shouldHide = isSender ? config.hideOASelfAvatar
                                  : config.hideOAOtherAvatar;
            break;

        default:
            return ((BOOL (*)(id, SEL))_orig_CommonMessageVM_isShowHeadImage)(self, _cmd);
    }

    if (shouldHide) {
        return NO;  // 隐藏头像
    }

    return ((BOOL (*)(id, SEL))_orig_CommonMessageVM_isShowHeadImage)(self, _cmd);
}

// ============================================================
// MARK: - Install
// ============================================================

@implementation HideAvatarHook

+ (void)install {
    WPLog(@"HideAvatar", @"HideAvatarHook install (复刻微信优化 1.6.5 isShowHeadImage)");
    Class cls;

    // ① Hook BaseMsgContentViewController.viewWillAppear: — 聊天类型分类
    cls = objc_getClass("BaseMsgContentViewController");
    if (cls) {
        MSHookMessageEx(cls,
            @selector(viewWillAppear:),
            (IMP)hook_viewWillAppear,
            &_orig_BaseMsgContentVC_viewWillAppear);
        WPLog(@"HideAvatar", @"[Hook] ✓ BaseMsgContentViewController.viewWillAppear:");
    } else {
        WPLog(@"HideAvatar", @"[Hook] ✗ BaseMsgContentViewController class not found");
    }

    // ② Hook CommonMessageViewModel.isShowHeadImage — 核心决策
    cls = objc_getClass("CommonMessageViewModel");
    if (cls) {
        MSHookMessageEx(cls,
            NSSelectorFromString(@"isShowHeadImage"),
            (IMP)hook_isShowHeadImage,
            &_orig_CommonMessageVM_isShowHeadImage);
        WPLog(@"HideAvatar", @"[Hook] ✓ CommonMessageViewModel.isShowHeadImage");
    } else {
        WPLog(@"HideAvatar", @"[Hook] ✗ CommonMessageViewModel class not found");
    }

    WPLog(@"HideAvatar", @"HideAvatarHook install complete");
}

@end
```

## 实现原理速查

```
进入聊天页面
    │
    ▼
BaseMsgContentViewController.viewWillAppear:  ← Hook
    │  获取 m_contact.m_nsUsrName
    │  gh_ → 公众号 | @chatroom → 群聊 | 其他 → 私聊
    │  更新全局 _currentChatType
    │
    ▼
消息 Cell 渲染 (每条消息触发)
    │
    ▼
CommonMessageViewModel.isShowHeadImage  ← Hook
    │  读取 _currentChatType
    │  读取 [viewModel isSender]
    │  (类型, isSender) → 查表 → PluginConfig 对应属性
    │  属性为 YES → return NO (隐藏)
    │  属性为 NO  → return 原始值
    │
    ▼
微信原生布局:
    isShowHeadImage=NO → 隐藏 MMHeadImageView, 气泡自动靠边
    isShowHeadImage=YES → 显示头像, 气泡在头像旁边
```

## 配置流向

```
SettingAvatarHideController 开关
    │  addSwitchRow key: "HideC2COtherAvatar" 等
    │  block 回调 → config.xxx = isOn → [config save]
    │
    ▼
PluginConfig.save
    │  [NSUserDefaults setBool: forKey:@"WCP_HideC2COtherAvatar"]
    │  [NSUserDefaults synchronize]
    │
    ▼
PluginConfig.loadDefaults (启动时)
    │  [NSUserDefaults boolForKey:@"WCP_HideC2COtherAvatar"]
    │  → config.hideC2COtherAvatar
    │
    ▼
HideAvatarHook.hook_isShowHeadImage
    │  读取 [PluginConfig shared].hideC2COtherAvatar
    │  → 决定返回 YES/NO
```

## MRC 注意事项

1. `_cachedUsername` 使用 `copy`/`release` 手动管理，**不要用 `retain`** 因为它是 `NSString *`
2. `PluginConfig` 单例通过 `[PluginConfig shared]` 获取，返回的是 autoreleased 对象，**不需要手动 release**
3. Hook 安装函数中使用的 `objc_getClass` 返回的 `Class` 是全局对象，不需要 retain/release
4. `WPLog` 宏内部使用了 `NSString stringWithFormat:`，返回 autoreleased 对象，OK
5. **不要使用 `__weak`、`__strong`、`__block` 等 ARC 修饰符** — MRC 下这些会导致编译错误

## 与微信优化的差异

| 项目 | 微信优化 | 本项目 |
|------|---------|--------|
| 聊天类型判定时机 | viewDidLoad + viewWillAppear | 仅 viewWillAppear |
| NSUserDefaults Suite | 独立 Suite `com.cyansmoke.wechattweak` | `standardUserDefaults` + `WCP_` 前缀 |
| 配置 key 命名 | `hideOtherAvatarInPrivateChat` | `WCP_HideC2COtherAvatar` |
| NSUserDefaults 获取 | 懒加载单例 (FUN_00007a88) | `[NSUserDefaults standardUserDefaults]` 直接使用 |
| Hook 安装时机 | 构造函数自动执行 | HookRegistry → FeatureModule → +install 链 |
| UI 框架 | 自定义 CSSettingItem + UISwitch | SettingCategoryController 基类封装 |

**功能等价**: 差异仅在于工程组织方式和命名风格，核心逻辑完全一致。两个 Hook 点和决策逻辑一行不差地复刻了反编译代码。
