#### 1.2.9 `computeLabelCenter:cViewFrame:avatarFrame:position:offsetX:offsetY:isSender:`

```objc
/// 根据位置编号计算 label.center（8 位置 + 偏移）
/// position:
///   0 = 头像上方, 1 = 头像下方, 2/7 = 消息旁边(气泡外),
///   3 = 消息下方(远离头像), 4 = 消息下方(靠近头像),
///   5 = 消息上方(远离头像), 6 = 消息上方(靠近头像)
static CGPoint computeLabelCenter(CGSize labelSize, CGRect cViewFrame, CGRect avatarFrame,
                                   NSInteger position, CGFloat offsetX, CGFloat offsetY,
                                   BOOL isSender) {
    CGPoint center = CGPointZero;
    CGFloat halfW = labelSize.width / 2;
    CGFloat halfH = labelSize.height / 2;

    switch (position) {
        case 0: // 头像上方
            center = CGPointMake(CGRectGetMidX(avatarFrame), CGRectGetMinY(avatarFrame) - halfH);
            break;
        case 1: // 头像下方
            center = CGPointMake(CGRectGetMidX(avatarFrame), CGRectGetMaxY(avatarFrame) + halfH);
            break;
        case 2: // 消息旁边(气泡外，发送方靠右，接收方靠左)
        case 7: // 同 case 2
            if (isSender) {
                center = CGPointMake(CGRectGetMaxX(cViewFrame) + halfW + 4, CGRectGetMidY(cViewFrame));
            } else {
                center = CGPointMake(CGRectGetMinX(cViewFrame) - halfW - 4, CGRectGetMidY(cViewFrame));
            }
            break;
        case 3: // 消息下方(远离头像)
            center = CGPointMake(CGRectGetMaxX(cViewFrame) - halfW, CGRectGetMaxY(cViewFrame) + halfH + 2);
            break;
        case 4: // 消息下方(靠近头像)
            center = CGPointMake(CGRectGetMinX(cViewFrame) + halfW, CGRectGetMaxY(cViewFrame) + halfH + 2);
            break;
        case 5: // 消息上方(远离头像)
            center = CGPointMake(CGRectGetMaxX(cViewFrame) - halfW, CGRectGetMinY(cViewFrame) - halfH - 2);
            break;
        case 6: // 消息上方(靠近头像)
            center = CGPointMake(CGRectGetMinX(cViewFrame) + halfW, CGRectGetMinY(cViewFrame) - halfH - 2);
            break;
        default: // 同 case 2/7
            if (isSender) {
                center = CGPointMake(CGRectGetMaxX(cViewFrame) + halfW + 4, CGRectGetMidY(cViewFrame));
            } else {
                center = CGPointMake(CGRectGetMinX(cViewFrame) - halfW - 4, CGRectGetMidY(cViewFrame));
            }
            break;
    }
    center.x += offsetX;
    center.y += offsetY;
    return center;
}
```

```objc
static void repl_CommonMessageCellView_updateNodeStatus(id self, SEL _cmd) {
    // 0. self 即为 cell（CommonMessageCellView 实例）
    id cell = self;

    // 1. 调用原方法
    ((void (*)(id, SEL, id))orig_CommonMessageCellView_updateNodeStatus)(self, _cmd, cell);

    // 2. 守卫：功能开关/VC 状态
    if (!isDarkOrNormalEnabled()) {
        hideTimeLabel(cell);
        return;
    }
    UILabel *label = getOrCreateTimeLabel(cell);
    if (!label) return;
    UIViewController *vc = [WPUtility findParentViewController:cell];
    if (vc.isBeingPresented || vc.isBeingDismissed) return;

    // 3. 数据提取
    id viewModel = getViewModelFromCell(cell);
    if (!viewModel) return;
    id wrap = getMessageWrapFromViewModel(viewModel);
    if (!wrap) return;

    // 4. 计算状态 + 格式化
    unsigned int createTime = getCreateTimeFromWrap(wrap);
    BOOL isSender = getIsSenderFromWrap(wrap);
    NSString *fromUsr = getFromUsrFromWrap(wrap);
    NSString *toUsr = getToUsrFromWrap(wrap);
    NSString *sessionKey = getSessionKeyForMessage(wrap, fromUsr, toUsr);
    unsigned int storedMax = getStoredMaxForSession(sessionKey, isSender);
    NSInteger statusCode = computeReadStatusCode(createTime, storedMax);

    NSString *labelText = formatMessageTime(createTime, statusCode);
    label.text = labelText;
    label.hidden = NO;

    // 5. 样式
    BOOL isDark = [WPUtility isDarkModeForViewController:vc];
    CGFloat fontSize = getTimeLabelFontSize(isDark, isSender);
    CGSize labelSize = computeLabelSize(labelText, fontSize, CGFLOAT_MAX);
    label.font = [UIFont systemFontOfSize:fontSize];
    label.frame = (CGRect){label.frame.origin, labelSize};
    label.textColor = getTimeLabelColor(label, isSender, isDark);
    label.layer.cornerRadius = getTimeLabelCornerRadius(isDark);

    // 6. 布局
    UIView *cv = getCellContentView(cell);
    CGRect cvFrame = getContentViewFrame(cell, cv);
    CGRect avatarFrame = getAvatarFrame(cell, cv);
    NSInteger position = getTimeLabelPosition(isSender);
    CGFloat offsetX = getTimeLabelOffsetX(isDark);
    CGFloat offsetY = getTimeLabelOffsetY(isDark);

    label.center = computeLabelCenter(labelSize, cvFrame, avatarFrame,
                                       position, offsetX, offsetY, isSender);
    addLabelToView(label, cv);
}
```

static void processRedEnvelopMessage(id self, SEL _cmd, id wrap) {
    // 守卫
    if (!isRedEnvelopEnabled()) return;
    if (!wrap) return;

    long long msgType = getMsgTypeFromWrap(wrap);
    if (msgType != 49) return;

    NSString *content = getContentFromWrap(wrap);
    if (![content containsString:@"wxpay://"]) return;

    // 去重
    NSString *msgId = getMsgIdFromWrap(wrap);
    if (isDuplicateRedEnvelop(msgId)) return;

    // 消息方向
    NSDictionary *contactInfo = getSelfContactInfo();
    NSString *selfUserName = contactInfo[@"userName"];
    NSString *fromUsr = getFromUsrFromWrap(wrap);
    NSString *toUsr = getToUsrFromWrap(wrap);
    NSDictionary *direction = determineMessageDirection(wrap, toUsr);

    // 开关 + 黑名单
    if (!shouldReceiveRedEnvelop(direction, selfUserName, fromUsr)) return;
    if (checkGroupBlacklist(toUsr)) return;

    // 解析
    id payInfoItem = getPayInfoItemFromWrap(wrap);
    NSDictionary *urlDict = parseRedEnvelopUrl(payInfoItem);
    if (!urlDict) return;

    // 参数构造
    id param = createRedEnvelopParam(urlDict, direction, selfUserName, fromUsr, toUsr, msgId);

    // 文本过滤
    if (applyTextFilter(content, [RedEnvelopConfig shared].redEnvelopTextFilter)) return;

    // 保存+调度
    NSString *sendId = urlDict[@"sendId"];
    savePendingRedEnvelop(param, sendId);
}

static void processTransferMessage(id wrap) {

- (BOOL)handleRevoke:(id)revokeWrap chatName:(NSString *)chatNameHint {
    // 守卫
    if (!revokeWrap || ![revokeWrap isKindOfClass:objc_getClass("CMessageWrap")]) {
        return [revokeWrap handleRevoke:revokeWrap chatName:chatNameHint]; // 调用原方法
    }
    if (isSelfRevoke(revokeWrap)) return [revokeWrap handleRevoke:revokeWrap chatName:chatNameHint];

    // XML 解析
    NSString *content = getContentFromWrap(revokeWrap);
    NSDictionary *revokeXml = parseRevokeXml(content);
    NSString *replacemsg = revokeXml[@"replacemsg"];
    if (!replacemsg) return [revokeWrap handleRevoke:revokeWrap chatName:chatNameHint];

    // 提取被撤回消息 ID
    NSString *revokedMsgId = getRevokedMessageId(revokeXml);
    if (!revokedMsgId) return [revokeWrap handleRevoke:revokeWrap chatName:chatNameHint];

    // 不提示模式
    if (revokeXml[@"noTip"]) return YES;

    // 查找原消息
    NSString *session = getSessionFromWrap(revokeWrap);
    id originalWrap = findOriginalMessage(revokedMsgId, session, chatNameHint);

    // 提取撤回者名字
    NSString *revokerName = extractRevokerName(replacemsg, revokeWrap);

    // 内容摘要
    NSString *digest = digestForMsgWrap(originalWrap);

    // 插入提示消息
    unsigned int createTime = getCreateTimeFromWrap(revokeWrap);
    NSString *tipContent = applyRevokeTemplate(revokerName, digest);
    id inserted = insertTipMessage_DKStyle(revokeWrap, tipContent, createTime);

    // 通知撤回者
    if (isNotifySenderEnabled(revokeWrap)) {
        NSString *notifyContent = getRevokeNotifyContent(revokeWrap,
            revokeXml[@"notifySenderTemplate"]);
        BOOL isGroup = isGroupSession(session);
        NSString *fromUsr = getFromUsrFromWrap(revokeWrap);
        NSString *toUsr = getToUsrFromWrap(revokeWrap);
        id notifyWrap = createNotifyMessageWrap(revokeWrap, notifyContent,
                                                  isGroup, fromUsr, toUsr);
        sendNotifyMessage(notifyWrap, session);
    }

    return inserted;
}
