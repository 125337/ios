## 一、准备工作

### 1.1 项目环境

- **项目路径**：`/www/wwwroot/ios/WeChatPlugin/`
- **语言**：Objective-C
- **Hook 方式**：Runtime Method Swizzling
- **最低 iOS 版本**：iOS 13.0（颜色选择器需 iOS 14.0）

### 1.2 确定功能范围

| 功能 | 目标 |
|------|------|
| 聊天界面名字颜色 | 修改撤回消息中用户昵称的显示颜色 |
| 聊天界面时间颜色 | 修改撤回消息中时间文本的显示颜色 |
| 聊天界面内容颜色 | 修改撤回消息中内容文本的显示颜色 |
| 深色/浅色模式 | 根据不同界面模式自动切换颜色 |
| 设置页面 | 提供颜色选择器 + 开关控制 |

## 二、步骤一：定义默认颜色常量

### 操作文件

`WeChatPlugin/Config/Constants.h`

### 变更内容

在 `Constants.h` 中添加以下常量声明（放在 `#endif` 之前）：

```objc
// 新增：名字颜色相关默认常量
static NSString *const kDefaultNameColor        = @"#FA5151";   // 浅色模式名字颜色（微信红）
static NSString *const kDefaultTimeColor        = @"#FA5151";   // 浅色模式时间颜色
static NSString *const kDefaultContentColor     = @"#999999";   // 浅色模式内容颜色（灰色）
static NSString *const kDefaultDarkNameColor    = @"#FF6B6B";   // 深色模式名字颜色（浅红）
static NSString *const kDefaultDarkTimeColor    = @"#FF6B6B";   // 深色模式时间颜色
static NSString *const kDefaultDarkContentColor = @"#BBBBBB";   // 深色模式内容颜色（浅灰）
```

### 说明

- 默认颜色采用微信品牌红色 `#FA5151`，与微信原生撤回提示颜色一致
- 深色模式下使用稍亮的 `#FF6B6B`，保证在深色背景下的可读性
- 内容颜色使用灰色系，让名字和时间颜色更突出

### 3.5 在 PluginConfig.m 中实现颜色计算方法

```objc
#pragma mark - 颜色工具方法

- (UIColor *)colorFromHex:(NSString *)hex {
    if (!hex.length) return nil;
    NSString *clean = [hex stringByReplacingOccurrencesOfString:@"#" withString:@""];
    if (clean.length < 6) return nil;
    unsigned int r = 0, g = 0, b = 0, a = 255;
    [[NSScanner scannerWithString:[clean substringWithRange:NSMakeRange(0, 2)]] scanHexInt:&r];
    [[NSScanner scannerWithString:[clean substringWithRange:NSMakeRange(2, 2)]] scanHexInt:&g];
    [[NSScanner scannerWithString:[clean substringWithRange:NSMakeRange(4, 2)]] scanHexInt:&b];
    if (clean.length >= 8) {
        [[NSScanner scannerWithString:[clean substringWithRange:NSMakeRange(6, 2)]] scanHexInt:&a];
    }
    return [UIColor colorWithRed:r/255.0 green:g/255.0 blue:b/255.0 alpha:a/255.0];
}

- (NSString *)hexFromColor:(UIColor *)color {
    if (!color) return nil;
    CGFloat r, g, b, a;
    [color getRed:&r green:&g blue:&b alpha:&a];
    return [NSString stringWithFormat:@"#%02X%02X%02X",
            (int)(r * 255), (int)(g * 255), (int)(b * 255)];
}

- (BOOL)isDarkMode {
    if (@available(iOS 13.0, *)) {
        return UITraitCollection.currentTraitCollection.userInterfaceStyle == UIUserInterfaceStyleDark;
    }
    return NO;
}

- (UIColor *)nameColor {
    if (!_customColorsEnabled) {
        return [self colorFromHex:kDefaultNameColor];
    }
    return [self colorFromHex:[self isDarkMode] ? _darkNameColorHex : _nameColorHex]
        ?: [self colorFromHex:kDefaultNameColor];
}

- (UIColor *)timeColor {
    if (!_customColorsEnabled) {
        return [self colorFromHex:kDefaultTimeColor];
    }
    return [self colorFromHex:[self isDarkMode] ? _darkTimeColorHex : _timeColorHex]
        ?: [self colorFromHex:kDefaultTimeColor];
}

- (UIColor *)contentColor {
    if (!_customColorsEnabled) {
        return [self colorFromHex:kDefaultContentColor];
    }
    return [self colorFromHex:[self isDarkMode] ? _darkContentColorHex : _contentColorHex]
        ?: [self colorFromHex:kDefaultContentColor];
}
```

```objc
#import <Foundation/Foundation.h>

/// 名字颜色 Hook：修改微信聊天界面撤回消息中昵称、时间、内容的显示颜色
@interface TintHook : NSObject
+ (void)install;
@end
```

---

### 4.2 创建 TintHook.m — 完整实现

```objc
#import "TintHook.h"
#import "../../Config/PluginConfig.h"
#import "../../Core/HookEngine.h"
#import <objc/runtime.h>
#import <objc/message.h>

// ============================================================
// 核心：对单个 Cell 执行颜色替换
// ============================================================
static void tintRevokeCell(id cell) {
    @try {
        if (!cell) return;

        // 第1步：获取消息包装对象 CMessageWrap
        id wrap = nil;
        for (NSString *sel in @[@"getCurrentMessageWrap", @"messageWrap"]) {
            SEL s = NSSelectorFromString(sel);
            if ([cell respondsToSelector:s]) {
                wrap = ((id (*)(id, SEL))objc_msgSend)(cell, s);
                break;
            }
        }
        Class CMessageWrapClass = objc_getClass("CMessageWrap");
        if (!wrap || (CMessageWrapClass && ![wrap isKindOfClass:CMessageWrapClass])) return;

        // 第2步：获取消息内容文本
        NSString *content = nil;
        if ([wrap respondsToSelector:@selector(m_nsContent)])
            content = ((id (*)(id, SEL))objc_msgSend)(wrap, @selector(m_nsContent));

        // 第3步：仅处理撤回消息
        if (![content containsString:@"撤回"]) return;

        PluginConfig *config = [PluginConfig shared];

        // 第4步：解析消息格式 "HH:mm:ss\n"xxx"撤回了一条消息"
        NSRange nl = [content rangeOfString:@"\n"];
        if (nl.location == NSNotFound || nl.location == 0) return;
        NSRange timeRange = NSMakeRange(0, nl.location);  // 时间行

        // 第5步：获取 RichTextView
        id rtv = [cell respondsToSelector:@selector(getRichTextView)]
            ? ((id (*)(id, SEL))objc_msgSend)(cell, @selector(getRichTextView))
            : nil;
        if (!rtv) @try { rtv = [cell valueForKey:@"m_richTextView"]; }
        @catch (NSException *e) {}

        // 第6步：修改 AttributedText 颜色
        if ([rtv respondsToSelector:@selector(attributedText)] &&
            [rtv respondsToSelector:@selector(setAttributedText:)]) {

            id attr = ((id (*)(id, SEL))objc_msgSend)(rtv, @selector(attributedText));
            NSMutableAttributedString *ma = [attr isKindOfClass:[NSAttributedString class]]
                ? [[NSMutableAttributedString alloc] initWithAttributedString:attr]
                : [[NSMutableAttributedString alloc] initWithString:content];

            if (NSMaxRange(timeRange) <= ma.length) {
                // 设置时间颜色
                [ma addAttribute:NSForegroundColorAttributeName
                           value:[config timeColor]
                           range:timeRange];

                // 设置名字颜色：在正文区域搜索 "" 包裹的名字
                NSRange contentStart = NSMakeRange(nl.location + 1,
                    ma.length - nl.location - 1);
                if (NSMaxRange(contentStart) <= ma.length && contentStart.length > 0) {
                    NSRange nameSearch = [ma.string rangeOfString:@"\""
                                                         options:0
                                                           range:contentStart];
                    if (nameSearch.location != NSNotFound) {
                        NSRange nameEnd = [ma.string rangeOfString:@"\""
                                                          options:0
                                                            range:NSMakeRange(nameSearch.location + 1,
                                                                ma.length - nameSearch.location - 1)];
                        if (nameEnd.location != NSNotFound) {
                            NSRange nameRange = NSMakeRange(nameSearch.location,
                                nameEnd.location - nameSearch.location + 1);
                            if (NSMaxRange(nameRange) <= ma.length) {
                                [ma addAttribute:NSForegroundColorAttributeName
                                           value:[config nameColor]
                                           range:nameRange];
                            }
                        }
                    }
                }

                // 应用修改
                ((void (*)(id, SEL, id))objc_msgSend)(rtv, @selector(setAttributedText:), ma);
            }
            [ma release];
        }

        // 第7步：触发重绘
        if ([rtv respondsToSelector:@selector(setNeedsDisplay)])
            ((void (*)(id, SEL))objc_msgSend)(rtv, @selector(setNeedsDisplay));

    } @catch (NSException *e) {}
}

// ============================================================
// 对指定微信 Cell 类的两个关键方法进行 Hook
// ============================================================
static NSMutableDictionary<NSString *, NSValue *> *gTintOrigIMPs = nil;

static void hookCellColor(NSString *className) {
    Class cls = objc_getClass(className.UTF8String);
    if (!cls) return;

    static dispatch_once_t onceToken;
    dispatch_once(&onceToken, ^{
        gTintOrigIMPs = [[NSMutableDictionary dictionary] retain];
    });

    for (NSString *methodName in @[@"updateStatus", @"layoutContentView"]) {
        SEL sel = NSSelectorFromString(methodName);
        Method m = class_getInstanceMethod(cls, sel);
        if (!m) continue;

        NSString *key = [NSString stringWithFormat:@"Tint_%@_%@", className, methodName];
        IMP origIMP = method_getImplementation(m);
        [gTintOrigIMPs setObject:[NSValue valueWithPointer:origIMP] forKey:key];

        IMP newIMP = imp_implementationWithBlock(^void(id self) {
            // 先调用原始实现（保证正常布局）
            NSValue *impValue = gTintOrigIMPs[key];
            if (impValue) {
                IMP orig = [impValue pointerValue];
                ((void (*)(id, SEL))orig)(self, sel);
            }
            // 再应用自定义颜色
            tintRevokeCell(self);
        });

        method_setImplementation(m, newIMP);
    }
}

// ============================================================
// install 入口：Hook 微信的四种消息 Cell 视图
// ============================================================
@implementation TintHook

+ (void)install {
    for (NSString *cls in @[
        @"SystemMessageCellView",   // 系统消息（撤回提示在此类显示）
        @"TextMessageCellView",     // 文本消息
        @"CommonMessageCellView",   // 通用消息
        @"BaseMessageCellView"      // 基类消息（兜底）
    ]) {
        hookCellColor(cls);
    }
}

@end
```

### 4.3 实现要点说明

| 步骤 | 说明 |
|------|------|
| 获取 CMessageWrap | 通过 `getCurrentMessageWrap` 或 `messageWrap` 方法获取消息数据对象 |
| 解析消息格式 | 微信撤回消息格式为 `"HH:mm:ss\n"用户昵称"撤回了一条消息"`，用 `\n` 分割时间和内容 |
| 定位名字 | 在内容区搜索第一个 `"` 到第二个 `"` 之间的范围，即为用户昵称 |
| 颜色替换 | 使用 `NSAttributedString` 的 `NSForegroundColorAttributeName` 属性设置颜色 |
| 先原始后自定义 | Block 中先执行原始 IMP 再执行颜色修改，保证微信原生逻辑不受影响 |


2. 在 `allModules` 方法中将 `TintHook` 添加到 `layout` 模块的 `hookInstallerClasses` 中：

```objc
// 修改前：
hookInstallerClasses:@[[UIPurifyHook class], [UISimplifyHook class], ...]
```

### 说明

`HookRegistry` 在启动时会遍历所有模块的 `hookInstallerClasses`，调用每个 Hook 类的 `+install` 方法。将 `TintHook` 注册到 `layout`（界面布局）模块中，因为它属于界面外观定制功能。

```objc
static NSString *keyForTag(NSInteger tag) {
    switch (tag) {
        case 2001: return @"ChatNameFontSize";
        default: return nil;
    }
}

static NSString *fontSizeForKey(NSString *key) {
    PluginConfig *c = [PluginConfig shared];
    CGFloat val = 0;
    if ([key isEqualToString:@"ChatNameFontSize"]) val = c.chatNameFontSize;
    return [NSString stringWithFormat:@"%.0fpt", val];
}

@implementation SettingNameColorController

- (void)viewDidLoad {
    [super viewDidLoad];
    self.title = @"名字颜色";
    [self buildUI];
}

#pragma mark - 字体大小输入弹窗

- (void)onNumericRowTap:(UIButton *)sender {
    NSString *key = keyForTag(sender.tag);
    if (!key) return;

    NSDictionary *configMap = @{
        @"ChatNameFontSize": @{
            @"title": @"字体大小",
            @"desc": @"请输入字体大小(10pt-24pt)",
            @"placeholder": @"16"
        },
    };

    NSDictionary *cfg = configMap[key];
    UIAlertController *alert = [UIAlertController
        alertControllerWithTitle:cfg[@"title"]
        message:cfg[@"desc"]
        preferredStyle:UIAlertControllerStyleAlert];

    [alert addTextFieldWithConfigurationHandler:^(UITextField *textField) {
        CGFloat val = [PluginConfig shared].chatNameFontSize;
        if (val == 0) val = 16;
        textField.placeholder = cfg[@"placeholder"];
        textField.text = [NSString stringWithFormat:@"%.0f", val];
        textField.keyboardType = UIKeyboardTypeNumbersAndPunctuation;
    }];

    [alert addAction:[UIAlertAction actionWithTitle:@"确定"
        style:UIAlertActionStyleDefault
        handler:^(UIAlertAction *action) {
            UITextField *tf = alert.textFields.firstObject;
            CGFloat val = [tf.text floatValue];
            if (val < 10) val = 10;
            if (val > 24) val = 24;
            [PluginConfig shared].chatNameFontSize = val;
            [[PluginConfig shared] save];
            [self buildUI];
        }]];

    [alert addAction:[UIAlertAction actionWithTitle:@"取消"
        style:UIAlertActionStyleCancel handler:nil]];

    [self presentViewController:alert animated:YES completion:nil];
}

#pragma mark - Build UI

- (void)buildUI {
    for (UIView *v in self.contentView.subviews) {
        [v removeFromSuperview];
    }
    [self.inputFields removeAllObjects];

    PluginConfig *config = [PluginConfig shared];

    UIView *card1 = [self addTableGroupAtY:y width:w];

        title:@"启用聊天界面名称颜色"

            *ecy = [self addColorRowInGroup:expand
                title:@"浅色模式文字颜色"
                key:@"nameColorHex"
                value:config.nameColorHex
                cy:*ecy width:w];
            *ecy = [self addSeparatorInGroup:expand cy:*ecy width:w];
            *ecy = [self addColorRowInGroup:expand
                title:@"深色模式文字颜色"
                key:@"darkNameColorHex"
                value:config.darkNameColorHex
                cy:*ecy width:w];
        } cy:cy1 width:w];

    cy1 = [self addSeparatorInGroup:card1 cy:cy1 width:w];

    // 2. 启用聊天界面名称大小 + 子功能

        title:@"启用聊天界面名称大小"

    y = [self finishGroup:card1 atY:y height:cy1];

    self.scrollView.contentSize = CGSizeMake(w, y + 40);
```

```objc
- (void)openNameColor:(id)sender {
    // 获取当前 ViewController
    UIResponder *responder = (UIResponder *)sender;
    while (responder) {
        if ([responder isKindOfClass:[UIViewController class]]) break;
        responder = [responder nextResponder];
    }
    UIViewController *vc = (UIViewController *)responder;
    if (!vc) return;

    // 动态创建 SettingNameColorController 并 Push
    Class cls = NSClassFromString(@"SettingNameColorController");
    if (!cls) return;
    UIViewController *subVC = [[cls alloc] init];
    if (subVC) {
        [vc.navigationController pushViewController:subVC animated:YES];
        [subVC release];
    }
}
```
