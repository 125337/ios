```objc
static IMP _orig_WCTableViewCellLeftConfig_title = NULL;

static id hook_WCTableViewCellLeftConfig_title(id self, SEL _cmd) {
    id orig = ((id (*)(id, SEL))_orig_WCTableViewCellLeftConfig_title)(self, _cmd);
    if (!SimplifyEnabled() || !orig) return orig;

    id repl = MenuNames()[orig];
    if (repl) return repl;

    // 处理复合标题如 "服务\n支付" → 取 \n 后的部分查字典
    NSRange nlRange = [orig rangeOfString:@"\n"];
    if (nlRange.location != NSNotFound) {
        NSString *suffix = [orig substringFromIndex:nlRange.location + 1];
        id subRepl = MenuNames()[suffix];
        if (subRepl) return subRepl;
    }
    return orig;
}
```

### 2.8 策略B ⭐ — MMUILabel.setText: (参照 FUN_00044a80, L40388)

**4 层上下文匹配决策树**，是整个功能最核心的 Hook：

```objc
static IMP _orig_MMUILabel_setText = NULL;

static void hook_MMUILabel_setText(id self, SEL _cmd, NSString *text) {
    // 【守卫1】nil 或未启用 → 调用原始 (L40409)
    if (!text || !SimplifyEnabled()) {
        ((void (*)(id, SEL, id))_orig_MMUILabel_setText)(self, _cmd, text);
        return;
    }

    // 【守卫2】白名单: "[" 开头的是 badge 数字，不处理 (L40413: cf__O_ = "[")
    if ([text hasPrefix:@"["]) {
        ((void (*)(id, SEL, id))_orig_MMUILabel_setText)(self, _cmd, text);
        return;
    }

    PluginConfig *cfg = [PluginConfig shared];

    // 【匹配层1】NavigationBar 上下文 → mainTitleReplacement (L40450-40473)
    // 沿 nextResponder 链向上查找包含 "NavigationBar" 的类
    if (cfg.mainTitleReplacement && cfg.mainTitleReplacement.length > 0) {
        id responder = self;
        while ((responder = [responder nextResponder])) {
            NSString *className = NSStringFromClass([responder class]);
            if ([className containsString:@"NavigationBar"]) {
                ((void (*)(id, SEL, id))_orig_MMUILabel_setText)(self, _cmd, cfg.mainTitleReplacement);
                return;
            }
        }
    }

    // 【匹配层2】Contacts 上下文 → contactsReplacement (L40475-40483)
    if (cfg.contactsReplacement && cfg.contactsReplacement.length > 0) {
        id responder = self;
        while ((responder = [responder nextResponder])) {
            NSString *className = NSStringFromClass([responder class]);
            if ([className containsString:@"Contacts"] || [className containsString:@"Contact"]) {
                if ([text hasPrefix:@"通讯录"] || [text isEqualToString:@"通讯录"]) {
                    ((void (*)(id, SEL, id))_orig_MMUILabel_setText)(self, _cmd, cfg.contactsReplacement);
                    return;
                }
            }
        }
    }

    // 【匹配层3】Discover 上下文 → discoverReplacement (L40486-40518)
    // 反编译中使用 cf_Ss 检查 "发现" 相关
    if (cfg.discoverReplacement && cfg.discoverReplacement.length > 0) {
        if ([text hasPrefix:@"发现"] || [text isEqualToString:@"发现"]) {
            ((void (*)(id, SEL, id))_orig_MMUILabel_setText)(self, _cmd, cfg.discoverReplacement);
            return;
        }
    }

    // 【匹配层4】好友数格式替换 (L40492-40511)
    // 正则匹配 \d+ 且在 "位联系人" 上下文中
    if (cfg.friendsCountReplacement && cfg.friendsCountReplacement.length > 0) {
        NSRegularExpression *regex = [NSRegularExpression
            regularExpressionWithPattern:@"\\d+" options:0 error:nil];
        NSTextCheckingResult *match = [regex firstMatchInString:text
            options:0 range:NSMakeRange(0, text.length)];
        if (match && [text containsString:@"位"] && [text containsString:@"联系人"]) {
            NSString *number = [text substringWithRange:match.range];
            NSString *formatted = [NSString stringWithFormat:cfg.friendsCountReplacement, number];
            ((void (*)(id, SEL, id))_orig_MMUILabel_setText)(self, _cmd, formatted);
            return;
        }
    }

    // 【默认】调用原始 (L40513)
    ((void (*)(id, SEL, id))_orig_MMUILabel_setText)(self, _cmd, text);
}
```

### 2.11 策略C — 钱包余额替换

```objc
// WCPayWalletEntryHeaderView 4个方法的共享逻辑 (参照 FUN_00045788/8a8/9c8/c00)
static void applyWalletBalance(id self) {
    NSString *replacement = [PluginConfig shared].walletBalanceReplacement;
    if (!replacement || replacement.length == 0) return;

    id timeoutNumber = nil;
    @try { timeoutNumber = [self valueForKey:@"_timeoutNumber"]; }
    @catch (NSException *e) {}

    if (timeoutNumber) {
        NSScanner *scanner = [NSScanner scannerWithString:replacement];
        double value;
        if ([scanner scanDouble:&value] && scanner.isAtEnd) {
            // 数字模式: 元→分 ×100 (参照 L40823)
            long cents = (long)(value * 100);
            [timeoutNumber performSelector:@selector(setText:)
                               withObject:[NSString stringWithFormat:@"%ld", cents]];
        } else {
            // 文本模式: 直接截取 (参照 L40818)
            long len = MIN(replacement.length, 10);
            [timeoutNumber performSelector:@selector(setText:)
                               withObject:[replacement substringToIndex:len]];
        }
    }

    // 同时更新余额标签 _balanceMoneyLabel (参照 L40918-40934, 仅 updateBalance 系列)
    // 正则替换数字部分
    id balanceLabel = nil;
    @try { balanceLabel = [self valueForKey:@"_balanceMoneyLabel"]; }
    @catch (NSException *e) {}
    if (balanceLabel && [balanceLabel respondsToSelector:@selector(text)]) {
        NSString *origText = [balanceLabel performSelector:@selector(text)];
        if (origText) {
            NSRegularExpression *regex = [NSRegularExpression
                regularExpressionWithPattern:@"\\d+(\\.\\d+)?" options:0 error:nil];
            NSString *newText = [regex stringByReplacingMatchesInString:origText
                options:0 range:NSMakeRange(0, origText.length) withTemplate:replacement];
            [balanceLabel performSelector:@selector(setText:) withObject:newText];
        }
    }
}

// 通过 nextResponder 链确认父级是 WCPayWalletEntryHeaderView

    if (SimplifyEnabled()) {
        NSString *replacement = [PluginConfig shared].walletBalanceReplacement;
        if (replacement && replacement.length > 0) {
```

```objc
            (IMP)hook_WCPay_updateBalanceEntryView, (IMP *)&_orig_WCPay_updateEntry);
        MSHookMessageEx(walletHeader, sel_registerName("updateBalanceAndRefreshView"),
            (IMP)hook_WCPay_updateBalanceAndRefreshView, (IMP *)&_orig_WCPay_updateRefresh);
    } else {
        WPLog(@"UISimplify", @"[Hook] - WCPayWalletEntryHeaderView not found");
    }

    Class timeoutNum = objc_getClass("TimeoutNumber");
    if (timeoutNum) {
        MSHookMessageEx(timeoutNum, sel_registerName("updateNumber:"),
            (IMP)hook_TimeoutNumber_updateNumber, (IMP *)&_orig_TimeoutNumber_updateNumber);
        MSHookMessageEx(timeoutNum, sel_registerName("defaultNumber:"),
            (IMP)hook_TimeoutNumber_defaultNumber, (IMP *)&_orig_TimeoutNumber_defaultNumber);
    }

    WPLog(@"UISimplify", @"UISimplifyHook install complete (15 hooks, 3 strategies)");
}
```
