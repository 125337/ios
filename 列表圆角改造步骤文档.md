## 1. 现状诊断 — MioPlugin 当前 Bug 清单

### 1.1 致命缺陷：Watchdog Timeout（已被禁用）

当前实现 `ListCornerRadiusHook.m` 已被**完全禁用**：

```objc
// Tweak.m:L6-L7
// ListCornerRadius disabled - will be re-enabled after research
// #import "Modules/ListCornerRadius/ListCornerRadiusHook.h"

// Tweak.m:L17
// [ListCornerRadiusHook initListCornerRadiusHook]; // disabled - causing watchdog timeout
```

**Watchdog timeout 的根本原因分析**：

问题出在 `cellPositionInSection` 函数中，`[tableView indexPathForCell:cell]` 调用发生在 `layoutSubviews` Hook 内，而这个调用**本身会触发 TableView 的布局更新**，形成 `layoutSubviews → indexPathForCell → 布局更新 → layoutSubviews` 的递归循环。

微信优化的做法是**先将 cell.superview 遍历到 UITableView**，再做 idx 查询，并且将所有 frame 修改放在索引查询**之后**，避免了在索引查询前修改 frame 导致触发布局。

#### 3.2 需要重写的主 Hook 函数 — 伪代码对照

**以下是改造后的函数必须实现的完整控制流**（与微信优化 FUN_0000b4cc 一一对应）：

```objc
static void replaced_MMTableViewCell_layoutSubviews(id self, SEL _cmd) {
    // ============ Step 1: 调用原始实现 ============
    if (_orig_MMTableViewCell_layoutSubviews) {
        ((void (*)(id, SEL))_orig_MMTableViewCell_layoutSubviews)(self, _cmd);
    }

    // ============ Step 2: 检查主开关 ============
    NSUserDefaults *defaults = [[NSUserDefaults alloc]
        initWithSuiteName:@"com.cyansmoke.wechattweak"];
    if (![defaults boolForKey:@"MoreDiscoverListRoundCornerEnabled"]) return;
    //                                               ↑ 改用这个 key

    // ============ Step 3: 通过响应链找 VC ============
    UIViewController *vc = findParentViewController((UIView *)self);
    if (!vc) return;
    NSString *className = NSStringFromClass([vc class]);

    // ============ Step 4: 暗黑模式判断 ============
    BOOL isDark = NO;
    if (@available(iOS 13.0, *)) {
        isDark = (vc.traitCollection.userInterfaceStyle == UIUserInterfaceStyleDark);
    }

    // ============ Step 5: 第1条链 — 背景色免设名单 (19个) ============
    static NSSet *bgSkipList = nil;
    static dispatch_once_t onceToken;
    dispatch_once(&onceToken, ^{
        bgSkipList = [NSSet setWithObjects:
            @"WCTimeLineViewController",
            @"WCAccountLoginUsersViewController",
            @"SessionSelectController",
            @"WCListViewController",
            @"BrandNotificationListViewController",
            @"BrandNewSessionViewController",
            @"BaseMsgContentViewController",
            @"BraceletRankProfileViewController",
            @"BraceletRankViewController",
            @"WCRedEnvelopesRedEnvelopesDetailViewController",
            @"MsgRecordDetailViewController",
            @"ChatRoomInfoViewController",
            @"ContactInfoViewController",
            @"AddFriendEntryViewController",
            @"AddContactToChatRoomViewController",
            @"SayHelloViewController",
            @"FTSHomeViewController",
            @"MMFinderPivotLiveViewController",
            @"WCSearchController",
            nil];
    });

    if (![bgSkipList containsObject:className]) {
        // VC 不在 19 个中 → 设置默认背景色
        ((UIView *)self).backgroundColor = WPCellDefaultBackgroundColor(isDark);
    }
    // VC 在 19 个中 → 不设背景色

    // ============ Step 6: FTSHome 单独判断 ============
    BOOL isFTSHome = [className isEqualToString:@"FTSHomeViewController"];
    // 此值将作为 flags 传入边框函数（FTSHome 跳过边框）

    // ============ Step 7: 第2条链 — 圆角排除名单 (18个, 不含FTSHome) ============
    static NSSet *cornerExcludeList = nil;
    dispatch_once(&onceToken, ^{
        cornerExcludeList = [NSSet setWithObjects:
            // 与 bgSkipList 相同，但不含 FTSHomeViewController
            @"WCTimeLineViewController",
            @"WCAccountLoginUsersViewController",
            @"SessionSelectController",
            @"WCListViewController",
            @"BrandNotificationListViewController",
            @"BrandNewSessionViewController",
            @"BaseMsgContentViewController",
            @"BraceletRankProfileViewController",
            @"BraceletRankViewController",
            @"WCRedEnvelopesRedEnvelopesDetailViewController",
            @"MsgRecordDetailViewController",
            @"ChatRoomInfoViewController",
            @"ContactInfoViewController",
            @"AddFriendEntryViewController",
            @"AddContactToChatRoomViewController",
            @"SayHelloViewController",
            @"MMFinderPivotLiveViewController",
            @"WCSearchController",
            nil];
    });

    if ([cornerExcludeList containsObject:className]) {
        return; // ★ VC 在 18 排除名单中 → 直接返回，不应用圆角
    }

    // ============ Step 8: 读取圆角半径和边距 ============
    NSInteger cornerRadius = [defaults integerForKey:@"MoreDiscoverListCornerRadius"];
    if (cornerRadius == 0) cornerRadius = 18;

    CGFloat screenWidth = [UIScreen mainScreen].bounds.size.width;

    NSInteger margin = [defaults integerForKey:@"CellHorizontalMargin"];
    if (margin == 0) margin = 9;

    // ============ Step 8.5: 判断是否 ContactsViewController ============
    BOOL isContacts = [className isEqualToString:@"ContactsViewController"];
    // (iVar4 变量)

    // ============ Step 9: 幂等 Frame 修改 ============
    UIView *cellView = (UIView *)self;
    CGFloat currentOriginX = cellView.frame.origin.x;
    UIView *superview = cellView.superview;
    CGFloat superviewOriginX = superview ? superview.frame.origin.x : 0;
    CGFloat targetX = (margin > superviewOriginX) ? margin - superviewOriginX : 0;

    // ★ 微信优化的安全检查条件，不是 != 比较
    if (currentOriginX - 2.0 * margin <= screenWidth) {
        CGRect frame = cellView.frame;
        frame.origin.x = targetX;  // 绝对值，非相对偏移
        cellView.frame = frame;
    }

    // ============ Step 10: 查找父级 TableView，获取 indexPath ============
    // ★ 关键修正：frame 修改在索引查询之前！
    UIView *parent = cellView.superview;
    UITableView *tableView = nil;
    while (parent) {
        if ([parent isKindOfClass:[UITableView class]]) {
            tableView = (UITableView *)parent;
            break;
        }
        parent = parent.superview;
    }
    if (!tableView) return;

    NSIndexPath *indexPath = [tableView indexPathForCell:(UITableViewCell *)self];
    if (!indexPath) return;

    NSInteger section = indexPath.section;
    NSInteger row = indexPath.row;
    NSInteger totalRows = [tableView numberOfRowsInSection:section];

    // ============ Step 11: 路径分叉 ============
    if (isContacts) {
        // ===== PATH B: 通讯录跨 Section 分组 =====
        [self wp_applyCornerForContacts:tableView
                              indexPath:indexPath
                                section:section
                                    row:row
                                  total:totalRows
                           cornerRadius:cornerRadius
                               isFTSHome:isFTSHome
                                defaults:defaults];
    } else {
        // ===== PATH A: 标准路径 =====
        [self wp_applyStandardCorner:tableView
                           indexPath:indexPath
                              section:section
                                  row:row
                                total:totalRows
                         cornerRadius:cornerRadius
                            isFTSHome:isFTSHome
                            className:className
                              defaults:defaults];
    }

    // ============ Step 12: clipsToBounds ============
    cellView.layer.masksToBounds = YES;
}
```

#### 3.3 需要新增的标准圆角方法（PATH A）

```objc
- (void)wp_applyStandardCorner:(UITableView *)tableView
                      indexPath:(NSIndexPath *)indexPath
                        section:(NSInteger)section
                            row:(NSInteger)row
                          total:(NSInteger)totalRows
                   cornerRadius:(NSInteger)configuredRadius
                      isFTSHome:(BOOL)isFTSHome
                      className:(NSString *)className
                        defaults:(NSUserDefaults *)defaults {

    UIView *cell = (UIView *)self;
    BOOL isNewMainFrame = [className isEqualToString:@"NewMainFrameViewController"];
    NSInteger indexSection = indexPath.section;  // 注意：这是 NSIndexPath section

    if (totalRows == 1) {
        // 唯一 Cell → 四角全圆
        cell.layer.cornerRadius = configuredRadius;
        cell.layer.maskedCorners = kCALayerMinXMinYCorner | kCALayerMaxXMinYCorner
                                 | kCALayerMinXMaxYCorner | kCALayerMaxXMaxYCorner;
        [self wp_applyBorderAndBg:cell radius:configuredRadius position:0 isFTSHome:isFTSHome defaults:defaults];

    } else if (row == 0) {
        // 首行 → 顶部两角
        cell.layer.cornerRadius = configuredRadius;
        cell.layer.maskedCorners = kCALayerMinXMinYCorner | kCALayerMaxXMinYCorner;
        [self wp_applyBorderAndBg:cell radius:configuredRadius position:1 isFTSHome:isFTSHome defaults:defaults];

    } else if (row == totalRows - 1) {
        // 末行 → NewMainFrame 折叠 Section 子判断
        BOOL isLastRowFullyRounded = YES;  // 默认执行底部两角

        if (isNewMainFrame && indexSection == 1 && totalRows == 1) {
            // 微信优化 L8487-8571: 折叠 Section 判断
            // 遍历 cell.subviews 找 MainFrameSectionFoldView
            MainFrameSectionFoldView *foldView = [self wp_findFoldViewInSubviews:cell.subviews];
            if (foldView && [foldView respondsToSelector:@selector(isFolding)]) {
                if ([foldView isFolding]) {
                    // isFolding == YES (已折叠) → 正常末行逻辑
                    // (保持 isLastRowFullyRounded = YES)
                } else {
                    // isFolding == NO (展开中) → 无圆角（中间行风格）
                    cell.layer.cornerRadius = 0;
                    cell.layer.maskedCorners = 0;
                    [self wp_applyBorderAndBg:cell radius:0 position:2 isFTSHome:isFTSHome defaults:defaults];
                    return;
                }
            }
        }

        // 底部两角
        cell.layer.cornerRadius = configuredRadius;
        cell.layer.maskedCorners = kCALayerMinXMaxYCorner | kCALayerMaxXMaxYCorner;
        [self wp_applyBorderAndBg:cell radius:configuredRadius position:3 isFTSHome:isFTSHome defaults:defaults];

    } else {
        // 中间行 → 无圆角
        cell.layer.cornerRadius = 0;
        cell.layer.maskedCorners = 0;
        [self wp_applyBorderAndBg:cell radius:0 position:2 isFTSHome:isFTSHome defaults:defaults];
    }
}

// 辅助：查找 MainFrameSectionFoldView
- (nullable UIView *)wp_findFoldViewInSubviews:(NSArray<UIView *> *)subviews {
    // 微信优化 L8493-8528: 遍历 subviews 找 MainFrameSectionFoldView
    for (UIView *subview in subviews) {
        if ([NSStringFromClass([subview class]) containsString:@"MainFrameSectionFoldView"]) {
            return subview;
        }
        // 递归子视图（微信优化用 while 循环遍历 superview 链）
        UIView *found = [self wp_findFoldViewInSubviews:subview.subviews];
        if (found) return found;
    }
    return nil;
}
```

#### 3.4 需要新增的通讯录分组方法（PATH B）

```objc
- (void)wp_applyCornerForContacts:(UITableView *)tableView
                         indexPath:(NSIndexPath *)indexPath
                           section:(NSInteger)section
                               row:(NSInteger)row
                             total:(NSInteger)rowInThisSection
                      cornerRadius:(NSInteger)radius
                          isFTSHome:(BOOL)isFTSHome
                           defaults:(NSUserDefaults *)defaults {

    UIView *cell = (UIView *)self;

    // 微信优化 L8586-8609: 遍历所有 Section 记录行数
    NSMutableArray<NSNumber *> *sectionRowCounts = [NSMutableArray array];
    NSInteger sectionCount = [tableView numberOfSections];
    NSInteger maxSections = MIN(sectionCount, 4);  // 微信优化限制最多4个 section
    for (NSInteger i = 0; i < maxSections; i++) {
        NSInteger rows = [tableView numberOfRowsInSection:i];
        [sectionRowCounts addObject:@(rows)];
    }

    // 找到第一个非空 Section 的首行
    NSInteger firstNonEmptySection = -1;
    for (NSInteger i = 0; i < sectionRowCounts.count; i++) {
        if ([sectionRowCounts[i] integerValue] > 0) {
            firstNonEmptySection = i;
            break;
        }
    }

    // 找到最后一个非空 Section 的末行
    NSInteger lastNonEmptySection = -1;
    for (NSInteger i = sectionRowCounts.count - 1; i >= 0; i--) {
        if ([sectionRowCounts[i] integerValue] > 0) {
            lastNonEmptySection = i;
            break;
        }
    }

    NSInteger sectionCount_effective = (NSInteger)sectionRowCounts.count;
    NSInteger firstSectionFirstRowCount = (firstNonEmptySection >= 0)
        ? [sectionRowCounts[firstNonEmptySection] integerValue] : 0;

    // 微信优化 L8616-8864: 跨 Section 分组逻辑
    if (sectionCount_effective == 1 && firstSectionFirstRowCount == 1) {
        // 仅一个 Section，仅一行 → 唯一 Cell
        cell.layer.cornerRadius = radius;
        cell.layer.maskedCorners = (kCALayerMinXMinYCorner | kCALayerMaxXMinYCorner
                                  | kCALayerMinXMaxYCorner | kCALayerMaxXMaxYCorner);
        [self wp_applyBorderAndBg:cell radius:radius position:0 isFTSHome:isFTSHome defaults:defaults];
    } else if (section == firstNonEmptySection && row == 0) {
        // 跨 Section 首行 → 顶部两角
        cell.layer.cornerRadius = radius;
        cell.layer.maskedCorners = kCALayerMinXMinYCorner | kCALayerMaxXMinYCorner;
        [self wp_applyBorderAndBg:cell radius:radius position:1 isFTSHome:isFTSHome defaults:defaults];
    } else if (section == lastNonEmptySection
               && row == [sectionRowCounts[lastNonEmptySection] integerValue] - 1) {
        // 跨 Section 末行 → 底部两角
        cell.layer.cornerRadius = radius;
        cell.layer.maskedCorners = kCALayerMinXMaxYCorner | kCALayerMaxXMaxYCorner;
        [self wp_applyBorderAndBg:cell radius:radius position:3 isFTSHome:isFTSHome defaults:defaults];
    } else {
        // 跨 Section 中间行 → 无圆角
        cell.layer.cornerRadius = 0;
        cell.layer.maskedCorners = 0;
        [self wp_applyBorderAndBg:cell radius:0 position:2 isFTSHome:isFTSHome defaults:defaults];
    }
}
```

```objc
// 微信优化 FUN_0000e344 等价逻辑
- (void)wp_applyBorderAndBg:(UIView *)cell
                     radius:(NSInteger)radius
                   position:(NSInteger)position
                  isFTSHome:(BOOL)isFTSHome
                   defaults:(NSUserDefaults *)defaults {

    // 1. 边框开关检查
    BOOL borderEnabled = [defaults boolForKey:@"CellBorderEnabled"];
    if (!borderEnabled || isFTSHome) {
        // isFTSHome → 跳过边框（微信优化 L9893-9894）
        return;
    }

    // 2. 边框宽度
    CGFloat borderWidth = [defaults floatForKey:@"CellBorderWidth"];
    if (borderWidth == 0) borderWidth = 2.0;

    // 3. 边框颜色 — 暗黑模式双色
    UIColor *borderColor = [self wp_loadDynamicColorLightKey:@"CellBorderColorLight"
                                                     darkKey:@"CellBorderColorDark"
                                                   defaults:defaults];

    // 4. 背景色
    UIColor *bgColor = [self wp_loadDynamicColorLightKey:@"CellBackgroundColorLight"
                                                 darkKey:@"CellBackgroundColorDark"
                                               defaults:defaults];
    if (bgColor) {
        cell.backgroundColor = bgColor;
    }

    // 5. 清理旧的 CAShapeLayer（Cell 复用时）
    NSArray *sublayers = [cell.layer.sublayers copy];
    for (CALayer *layer in sublayers) {
        if ([layer.name isEqualToString:@"com.mio.cornerBorder"]) {
            [layer removeFromSuperlayer];
        }
    }

    // 6. 按 position 构建边框
    switch (position) {
        case 0: {
            // 唯一Cell: 直接设置 layer 属性
            cell.layer.borderColor = borderColor.CGColor;
            cell.layer.borderWidth = borderWidth;
            break;
        }
        case 1: {
            // 首行: 顶部圆角 + 两侧直线
            CAShapeLayer *shape = [self wp_buildTopBorderPath:cell.bounds
                                                  borderWidth:borderWidth
                                                  borderColor:borderColor
                                                       radius:radius];
            shape.name = @"com.mio.cornerBorder";
            [cell.layer addSublayer:shape];
            break;
        }
        case 2: {
            // 中间行: 仅左右两侧直线
            CAShapeLayer *left  = [self wp_buildLinePath:cell.bounds
                                             borderWidth:borderWidth
                                             borderColor:borderColor
                                                    side:@"left"];
            CAShapeLayer *right = [self wp_buildLinePath:cell.bounds
                                             borderWidth:borderWidth
                                             borderColor:borderColor
                                                    side:@"right"];
            left.name = @"com.mio.cornerBorder";
            right.name = @"com.mio.cornerBorder";
            [cell.layer addSublayer:left];
            [cell.layer addSublayer:right];
            break;
        }
        case 3: {
            // 末行: 底部圆角 + 两侧直线
            CAShapeLayer *shape = [self wp_buildBottomBorderPath:cell.bounds
                                                     borderWidth:borderWidth
                                                     borderColor:borderColor
                                                          radius:radius];
            shape.name = @"com.mio.cornerBorder";
            [cell.layer addSublayer:shape];
            break;
        }
    }
}
```

```objc
static void (*_orig_UIView_layoutSubviews)(id, SEL);

static void _hooked_UIView_layoutSubviews(id self, SEL _cmd) {
    _orig_UIView_layoutSubviews(self, _cmd);

    // 1. 找父 VC
    UIViewController *vc = findParentViewController((UIView *)self);
    if (!vc) return;

    // 2. ★ 仅对 NewMainFrameViewController 生效
    if (![NSStringFromClass([vc class]) isEqualToString:@"NewMainFrameViewController"]) return;

    NSUserDefaults *d = [[NSUserDefaults alloc]
        initWithSuiteName:@"com.cyansmoke.wechattweak"];

    // 3. 主开关
    if (![d boolForKey:@"MoreDiscoverListRoundCornerEnabled"]) return;

    // 4. 读取参数
    NSInteger radius = [d integerForKey:@"MoreDiscoverListCornerRadius"];
    if (radius == 0) radius = 18;
    NSInteger margin = [d integerForKey:@"CellHorizontalMargin"];
    if (margin == 0) margin = 9;

    // 5. 修改 frame.origin.x
    UIView *view = (UIView *)self;
    CGFloat screenWidth = [UIScreen mainScreen].bounds.size.width;
    if (view.frame.origin.x - 2 * margin <= screenWidth) {
        CGRect frame = view.frame;
        frame.origin.x = margin;
        view.frame = frame;
    }

    // 6. 折叠判断
    if ([view respondsToSelector:@selector(isFolding)]) {
        if ([(id)view isFolding]) {
            // 已折叠 → 四角全圆
            view.layer.cornerRadius = radius;
            view.layer.maskedCorners = 0xF;
        } else {
            // 未折叠/展开中 → 底部两角
            view.layer.cornerRadius = radius;
            view.layer.maskedCorners = 0xC;
        }
    } else {
        // 无 isFolding → 四角全圆
        view.layer.cornerRadius = radius;
        view.layer.maskedCorners = 0xF;
    }

    // 7. 设置背景色
    view.backgroundColor = [WPCellDefaultBackgroundColor isDarkMode];
    view.layer.masksToBounds = YES;
}

void WPInstallSearchBoxHook(void) {
    MSHookMessageEx([UIView class], @selector(layoutSubviews),
        (IMP)_hooked_UIView_layoutSubviews, (IMP *)&_orig_UIView_layoutSubviews);
}
```
