# 好友检测功能 — 完全改造实施文档

> 从「本地属性检测」（不可靠）→「转账预下单检测」（精准，参照微信优化 1.6.5）

---

## 零、现状诊断：为什么现有方案不可靠

现有 `FriendDetectionHook.m` 使用本地属性检测法：

```objc
// 行 166-177 — 当前检测逻辑
BOOL isDeleted = !(contact && [contact isKindOfClass:cContactCls]);
if (!isDeleted) {
    unsigned int vf = [[contact valueForKey:@"m_uiVerifyFlag"] unsignedIntValue];
    if (vf > 0) { isDeleted = YES; }        // ❌ 很多被删好友 verifyFlag 仍为 0
    else {
        unsigned int status = [[contact valueForKey:@"m_uiStatus"] unsignedIntValue];
        if (status > 0) isDeleted = YES;    // ❌ m_uiStatus 不是好友关系字段
    }
}
```

**问题：**
- `m_uiVerifyFlag` 在本地缓存中不会实时更新，被删除后可能仍为 0
- `m_uiStatus` 不是好友关系状态字段
- 完全依赖本地数据库，被删好友的联系人记录可能仍存在于本地
- **必须通过服务端验证才能准确判断**

---

## 一、改造目标

将检测方式从"读本地联系人属性"改为"发转账预下单请求 → 拦截服务端返回的 retcode → 判断关系"。

与微信优化 1.6.5 的方案完全对齐：

| 对比维度 | 旧方案（本地属性） | 新方案（转账预下单） |
|---------|------------------|-------------------|
| 检测依据 | 本地 CContact 属性 | 服务端 retcode |
| 准确性 | 不可靠（本地缓存延迟） | 100% 准确 |
| 隐蔽性 | 无网络请求 | 预下单不通知对方 |
| 结果分类 | 删除/不删 | 正常 / 被删除 / 异常（封号） |

/*
 * 微信内部的转账预下单请求结构体
 * 通过 objc_msgSend 动态调用属性 setter，不需要 import 微信头文件
 *
 * 属性映射（反编译自 微信优化 行 18992-19042）：
 *   m_nsReceiverUserName  → setM_nsReceiverUserName:  // 对方微信号
 *   m_uiAmount            → setM_uiAmount:            // 金额（最小单位，1 = 0.01 元）
 *   m_uiFee               → setM_uiFee:               // 手续费
 *   m_uiFeeType           → setM_uiFeeType:           // 费用类型 = 31
 *   m_transferScene       → setM_transferScene:       // 转账场景 = 2
 *   m_uiPayScene          → setM_uiPayScene:          // 支付场景 = 11
 *   m_nsTraceInfo         → setM_nsTraceInfo:         // 追踪信息
 *   m_placeorderReserves  → setPlaceorderReserves:    // 预下单保留字段
 */
static id _Nullable createTransferRequest(NSString *receiverWxID) {
    Class reqCls = objc_getClass("WCPayTransferPrepayRequestStruct");
    if (!reqCls) {
        WPLog(@"FriendDetect", @"[TransferReq] WCPayTransferPrepayRequestStruct class not found");
        return nil;
    }

    // alloc + init
    id req = ((id (*)(Class, SEL))objc_msgSend)(reqCls, sel_registerName("alloc"));
    if (!req) return nil;
    req = ((id (*)(id, SEL))objc_msgSend)(req, @selector(init));
    if (!req) return nil;

    // 设置对方微信号
    SEL setReceiver = sel_registerName("setM_nsReceiverUserName:");
    if ([req respondsToSelector:setReceiver]) {
        ((void (*)(id, SEL, id))objc_msgSend)(req, setReceiver, receiverWxID);
    }

    // 设置金额 = 1（最小单位，对应 0.01 元）
    SEL setAmount = sel_registerName("setM_uiAmount:");
    if ([req respondsToSelector:setAmount]) {
        ((void (*)(id, SEL, unsigned int))objc_msgSend)(req, setAmount, 1);
    }

    // 设置手续费 = 1
    SEL setFee = sel_registerName("setM_uiFee:");
    if ([req respondsToSelector:setFee]) {
        ((void (*)(id, SEL, unsigned int))objc_msgSend)(req, setFee, 1);
    }

    // 设置费用类型 = 31
    SEL setFeeType = sel_registerName("setM_uiFeeType:");
    if ([req respondsToSelector:setFeeType]) {
        ((void (*)(id, SEL, unsigned int))objc_msgSend)(req, setFeeType, 31);
    }

    // 设置转账场景 = 2（特殊场景，用于好友检测）
    SEL setScene = sel_registerName("setM_transferScene:");
    if ([req respondsToSelector:setScene]) {
        ((void (*)(id, SEL, unsigned int))objc_msgSend)(req, setScene, 2);
    }

    // 设置支付场景 = 11
    SEL setPayScene = sel_registerName("setM_uiPayScene:");
    if ([req respondsToSelector:setPayScene]) {
        ((void (*)(id, SEL, unsigned int))objc_msgSend)(req, setPayScene, 11);
    }

    // 设置追踪信息 = "MioFriendDetector"
    SEL setTraceInfo = sel_registerName("setM_nsTraceInfo:");
    if ([req respondsToSelector:setTraceInfo]) {
        ((void (*)(id, SEL, id))objc_msgSend)(req, setTraceInfo, @"MioFriendDetector");
    }

    // 设置预下单保留字段 = 当前时间戳
    SEL setPlaceorderReserves = sel_registerName("setPlaceorderReserves:");
    if ([req respondsToSelector:setPlaceorderReserves]) {
        NSString *ts = [NSString stringWithFormat:@"%ld", (long)[[NSDate date] timeIntervalSince1970]];
        ((void (*)(id, SEL, id))objc_msgSend)(req, setPlaceorderReserves, ts);
    }

    WPLog(@"FriendDetect", @"[TransferReq] Created request for %@", receiverWxID);
    return req;
}

/**
 * 通过 WCPayLogicMgr 发送预下单请求
 *
 * 调用链（从反编译还原）：
 *   1. 获取 WCPayLogicMgr（通过 MMServiceCenter）
 *   2. 调用某个内部方法发送请求
 *
 * 注意：微信内部的具体发送方法名可能因版本不同而变化。
 * 以下提供一个基于反编译分析的调用路径作为主要尝试。
 */
static BOOL sendTransferPrepayRequest(id request) {
    // 获取 WCPayLogicMgr 实例
    id payMgr = WXGetService(objc_getClass("WCPayLogicMgr"));
    if (!payMgr) {
        WPLog(@"FriendDetect", @"[SendReq] WCPayLogicMgr not found");
        return NO;
    }

    // 尝试方案 A：直接调用 WCPayLogicMgr 的方法发送
    // 微信内部可能有类似 sendTransferPrepayRequest: 或 handleTransferPrepayRequest: 的方法
    // 根据反编译分析，微信优化是直接通过微信支付系统发送的

    // 方案 A-1：尝试通过 CContactMgr 获取联系人并发送
    // （这部分需要根据实际调试确定准确的调用路径）

    // 方案 A-2：通过 WCPayLogicMgr 的公开接口
    SEL sendSel = sel_registerName("onGetTransferPrepayRequest:");
    if ([payMgr respondsToSelector:sendSel]) {
        WPLog(@"FriendDetect", @"[SendReq] Trying onGetTransferPrepayRequest:");
        ((void (*)(id, SEL, id))objc_msgSend)(payMgr, sendSel, request);
        return YES;
    }

    // 方案 B：尝试通过 WCPayTransferPrepayRequestStruct 自身的 send 方法
    SEL reqSendSel = sel_registerName("startRequest");
    if ([request respondsToSelector:reqSendSel]) {
        WPLog(@"FriendDetect", @"[SendReq] Trying startRequest");
        ((void (*)(id, SEL))objc_msgSend)(request, reqSendSel);
        return YES;
    }

    // 方案 C：通过 MMServiceCenter 获取支付相关的 service
    // 微信支付的请求通常通过 Cgi 或 NetWork 层发送
    // 如果上述方法都不存在，可能需要通过 runtime 进一步探索

    WPLog(@"FriendDetect", @"[SendReq] WARNING: No suitable send method found");
    WPLog(@"FriendDetect", @"[SendReq] 请通过 FLEX/class-dump 确认 WCPayLogicMgr 的发送方法名");
    return NO;
}

/// Hook: WCPayLogicMgr - OnGetTransferPrepayRespone:
/// 微信服务端返回转账预下单结果时触发
static void hook_OnGetTransferPrepayRespone(id self, SEL _cmd, id response) {
    // ① 先调用原始实现
    if (_orig_OnGetTransferPrepayRespone) {
        ((void (*)(id, SEL, id))_orig_OnGetTransferPrepayRespone)(self, _cmd, response);
    }

    // ② 只在检测模式下处理
    if (!_detectionInProgress) return;
    if (!response) return;

    WPLog(@"FriendDetect", @"[Hook] OnGetTransferPrepayRespone called");

    // ③ 解析响应，提取 retcode / retmsg / wx_error_msg
    NSMutableDictionary *result = [NSMutableDictionary dictionary];

    if ([response isKindOfClass:[NSDictionary class]]) {
        // 响应已经是字典
        NSDictionary *dict = (NSDictionary *)response;
        result[@"retcode"] = dict[@"retcode"];
        result[@"retmsg"] = dict[@"retmsg"];
        result[@"wx_error_msg"] = dict[@"wx_error_msg"];
    } else {
        // 从对象中提取属性
        SEL retcodeSel = sel_registerName("retcode");
        if ([response respondsToSelector:retcodeSel]) {
            NSInteger rc = ((NSInteger (*)(id, SEL))objc_msgSend)(response, retcodeSel);
            result[@"retcode"] = @(rc);
        }

        SEL retmsgSel = sel_registerName("retmsg");
        if ([response respondsToSelector:retmsgSel]) {
            id msg = ((id (*)(id, SEL))objc_msgSend)(response, retmsgSel);
            result[@"retmsg"] = msg;
        }

        SEL errSel = sel_registerName("wx_error_msg");
        if ([response respondsToSelector:errSel]) {
            id err = ((id (*)(id, SEL))objc_msgSend)(response, errSel);
            result[@"wx_error_msg"] = err;
        }
    }

    WPLog(@"FriendDetect", @"[Hook] Response for %@: %@", _currentCheckingWxID, result);

    // ④ 如果是从检测循环发起的请求，存入全局变量并 signal
    @synchronized (result) {
        _currentCheckResponse = [result copy];
    }

    if (_detectionSemaphore) {
        dispatch_semaphore_signal(_detectionSemaphore);
        WPLog(@"FriendDetect", @"[Hook] Semaphore signaled");
    }
}

/// Hook: WCPayLogicMgr - insideCallBackGetTransferPrepayResponse:OnRequest:
/// 备用回调路径（部分微信版本可能走这个路径）
static void hook_insideCallback(id self, SEL _cmd, id response, id request) {
    // 先调用原始实现
    if (_orig_insideCallBackGetTransferPrepayResponse_OnRequest) {
        ((void (*)(id, SEL, id, id))_orig_insideCallBackGetTransferPrepayResponse_OnRequest)(self, _cmd, response, request);
    }

    // 如果在检测模式且主回调没有处理，这里作为兜底
    if (!_detectionInProgress) return;
    if (_currentCheckResponse) return; // 主回调已经处理过

    // 处理逻辑同 hook_OnGetTransferPrepayRespone
    if ([response isKindOfClass:[NSDictionary class]]) {
        _currentCheckResponse = [response copy];
    } else {
        NSMutableDictionary *result = [NSMutableDictionary dictionary];
        SEL retcodeSel = sel_registerName("retcode");
        if ([response respondsToSelector:retcodeSel]) {
            NSInteger rc = ((NSInteger (*)(id, SEL))objc_msgSend)(response, retcodeSel);
            result[@"retcode"] = @(rc);
        }
        _currentCheckResponse = [result copy];
    }

    if (_detectionSemaphore) {
        dispatch_semaphore_signal(_detectionSemaphore);
    }
}

/// 安装 WCPayLogicMgr 的 Hook（仅安装一次）
static void installPayLogicMgrHooks(void) {
    static dispatch_once_t onceToken;
    dispatch_once(&onceToken, ^{
        WPLog(@"FriendDetect", @"[HookInstall] Installing WCPayLogicMgr hooks...");

        Class payCls = objc_getClass("WCPayLogicMgr");
        if (!payCls) {
            WPLog(@"FriendDetect", @"[HookInstall] WCPayLogicMgr class not found");
            return;
        }

        // Hook 1: OnGetTransferPrepayRespone:
        SEL sel1 = sel_registerName("OnGetTransferPrepayRespone:");
        Method m1 = class_getInstanceMethod(payCls, sel1);
        if (m1) {
            MSHookMessageEx(payCls, sel1, (IMP)hook_OnGetTransferPrepayRespone, &_orig_OnGetTransferPrepayRespone);
            WPLog(@"FriendDetect", @"[HookInstall] ✓ Hooked OnGetTransferPrepayRespone:");
        } else {
            WPLog(@"FriendDetect", @"[HookInstall] ✗ OnGetTransferPrepayRespone: not found");
        }

        // Hook 2: insideCallBackGetTransferPrepayResponse:OnRequest:
        SEL sel2 = sel_registerName("insideCallBackGetTransferPrepayResponse:OnRequest:");
        Method m2 = class_getInstanceMethod(payCls, sel2);
        if (m2) {
            MSHookMessageEx(payCls, sel2, (IMP)hook_insideCallback, &_orig_insideCallBackGetTransferPrepayResponse_OnRequest);
            WPLog(@"FriendDetect", @"[HookInstall] ✓ Hooked insideCallBackGetTransferPrepayResponse:OnRequest:");
        } else {
            WPLog(@"FriendDetect", @"[HookInstall] ✗ insideCallBackGetTransferPrepayResponse:OnRequest: not found");
        }

        _hookInstalled = YES;
    });
}

/// 获取所有真实好友（排除群聊、公众号、自己）
static NSArray<NSDictionary *> *getAllFriends(void) {
    id contactMgr = WXGetService(objc_getClass("CContactMgr"));
    if (!contactMgr) {
        WPLog(@"FriendDetect", @"[getAllFriends] CContactMgr not found");
        return @[];
    }

    // 尝试多次参数获取联系人列表
    SEL sel = sel_registerName("getContactList:contactType:");
    if (![contactMgr respondsToSelector:sel]) {
        WPLog(@"FriendDetect", @"[getAllFriends] getContactList:contactType: not found");
        return @[];
    }

    NSArray *contacts = nil;
    // 尝试 (0, 8) — 最常见
    contacts = ((NSArray *(*)(id, SEL, int, int))objc_msgSend)(contactMgr, sel, 0, 8);
    if (!contacts || contacts.count == 0) {
        contacts = ((NSArray *(*)(id, SEL, int, int))objc_msgSend)(contactMgr, sel, 0, 0);
    }
    if (!contacts || contacts.count == 0) {
        contacts = ((NSArray *(*)(id, SEL, int, int))objc_msgSend)(contactMgr, sel, 1, 0);
    }
    if (!contacts || contacts.count == 0) {
        WPLog(@"FriendDetect", @"[getAllFriends] All methods returned nil");
        return @[];
    }

    // 获取自己的 wxid 以便排除
    NSString *selfWxID = nil;
    if ([contactMgr respondsToSelector:sel_registerName("getSelfContact")]) {
        id selfContact = ((id (*)(id, SEL))objc_msgSend)(contactMgr, sel_registerName("getSelfContact"));
        if ([selfContact respondsToSelector:sel_registerName("m_nsUsrName")]) {
            selfWxID = ((id (*)(id, SEL))objc_msgSend)(selfContact, sel_registerName("m_nsUsrName"));
        }
    }

    SEL wxidSel = sel_registerName("m_nsUsrName");
    SEL nickSel = sel_registerName("m_nsNickName");
    SEL remarkSel = sel_registerName("m_nsRemark");

    NSMutableArray *friends = [NSMutableArray array];
    for (id contact in contacts) {
        @try {
            NSString *wxID = ((id (*)(id, SEL))objc_msgSend)(contact, wxidSel);
            if (!wxID || wxID.length == 0) continue;
            // 排除群聊
            if ([wxID containsString:@"@chatroom"]) continue;
            // 排除公众号
            if ([wxID hasPrefix:@"gh_"]) continue;
            // 排除自己
            if (selfWxID && [wxID isEqualToString:selfWxID]) continue;

            NSString *nick = nil;
            if ([contact respondsToSelector:nickSel]) {
                nick = ((id (*)(id, SEL))objc_msgSend)(contact, nickSel);
            }

            NSString *remark = nil;
            if ([contact respondsToSelector:remarkSel]) {
                remark = ((id (*)(id, SEL))objc_msgSend)(contact, remarkSel);
            }

            [friends addObject:@{
                @"wxID": wxID,
                @"nickname": nick ?: @"",
                @"remark": remark ?: @"",
            }];
        } @catch (NSException *e) {
            WPLog(@"FriendDetect", @"[getAllFriends] Exception for contact: %@", e);
        }
    }

    WPLog(@"FriendDetect", @"[getAllFriends] Total: %lu friends (filtered)", (unsigned long)friends.count);
    return [friends copy];
}

/// 根据服务端返回的 retcode 判断好友状态
static MioFriendStatus determineStatusFromResponse(NSDictionary *response) {
    if (!response) return MioFriendStatusInvalid;

    NSInteger retcode = [response[@"retcode"] integerValue];
    NSString *retmsg = response[@"retmsg"];
    NSString *wxError = response[@"wx_error_msg"];

    // retcode == 0 通常表示成功，对方仍是好友
    if (retcode == 0) {
        return MioFriendStatusNormal;
    }

    // 被删除的常见错误码/错误信息
    // 微信服务端返回的典型错误：
    // - retcode != 0 且 retmsg 包含"不是好友"/"好友验证"等
    // - wx_error_msg 包含类似信息

    if (retmsg) {
        NSString *lowerMsg = [retmsg lowercaseString];
        if ([lowerMsg containsString:@"不是好友"] ||
            [lowerMsg containsString:@"好友验证"] ||
            [lowerMsg containsString:@"add friend"] ||
            [lowerMsg containsString:@"not friend"] ||
            [lowerMsg containsString:@"verify"]) {
            return MioFriendStatusDeleted;
        }
    }

    if (wxError) {
        NSString *lowerErr = [wxError lowercaseString];
        if ([lowerErr containsString:@"不是好友"] ||
            [lowerErr containsString:@"好友验证"] ||
            [lowerErr containsString:@"not friend"]) {
            return MioFriendStatusDeleted;
        }
    }

    // 其他非零 retcode → 可能是异常（账号异常、被封号等）
    WPLog(@"FriendDetect", @"[Determine] Unknown retcode=%ld retmsg=%@ wxError=%@ → Invalid",
          (long)retcode, retmsg, wxError);
    return MioFriendStatusInvalid;
}

- (void)startNewDetection:(MioFriendDetectionProgressBlock)progress
               completion:(MioFriendDetectionCompletionBlock)completion {
    if (self.isDetecting) {
        WPLog(@"FriendDetect", @"[StartNew] Already detecting, ignoring");
        if (completion) {
            dispatch_async(dispatch_get_main_queue(), ^{
                NSError *err = [NSError errorWithDomain:@"MioFriendDetect"
                                                   code:-1
                                               userInfo:@{NSLocalizedDescriptionKey: @"检测正在进行中"}];
                completion(nil, err);
            });
        }
        return;
    }

    // 安装 Hook（只安装一次）
    installPayLogicMgrHooks();
    if (!_hookInstalled) {
        WPLog(@"FriendDetect", @"[StartNew] Hook install failed");
        if (completion) {
            dispatch_async(dispatch_get_main_queue(), ^{
                NSError *err = [NSError errorWithDomain:@"MioFriendDetect"
                                                   code:-2
                                               userInfo:@{NSLocalizedDescriptionKey: @"Hook 安装失败，无法检测"}];
                completion(nil, err);
            });
        }
        return;
    }

    // 清除旧结果
    [MioFriendDetector clearSavedSummary];

    self.isDetecting = YES;
    self.shouldStop = NO;

    dispatch_async(dispatch_get_global_queue(DISPATCH_QUEUE_PRIORITY_DEFAULT, 0), ^{
        [self executeDetectionWithProgress:progress completion:completion resumeFrom:nil];
    });
}

                                                   code:-1
                                               userInfo:@{NSLocalizedDescriptionKey: @"检测正在进行中"}];

            });

    installPayLogicMgrHooks();
    if (!_hookInstalled) {

                completion(nil, [NSError errorWithDomain:@"MioFriendDetect"
                                                    code:-2
                                                userInfo:@{NSLocalizedDescriptionKey: @"Hook 安装失败"}]);
            });
        }
        return;
    }

    MioFriendDetectSummary *saved = [MioFriendDetector loadSavedSummary];
    if (!saved || !saved.checkedWxIDs || saved.checkedWxIDs.count == 0) {
        WPLog(@"FriendDetect", @"[Resume] No saved progress, starting new");
        [self startNewDetection:progress completion:completion];
        return;

    self.isDetecting = YES;
    self.shouldStop = NO;

    dispatch_async(dispatch_get_global_queue(DISPATCH_QUEUE_PRIORITY_DEFAULT, 0), ^{
        [self executeDetectionWithProgress:progress completion:completion resumeFrom:saved];
    });
}

#pragma mark - 核心检测循环

- (void)executeDetectionWithProgress:(MioFriendDetectionProgressBlock)progress
                          completion:(MioFriendDetectionCompletionBlock)completion
                          resumeFrom:(MioFriendDetectSummary *)resumeFrom {
    _detectionInProgress = YES;
    WPLog(@"FriendDetect", @"========================================");
    WPLog(@"FriendDetect", @" Detection started");
    WPLog(@"FriendDetect", @"========================================");

    // ① 获取所有好友
    NSArray<NSDictionary *> *allFriends = getAllFriends();
    if (allFriends.count == 0) {
        WPLog(@"FriendDetect", @"[Execute] No friends found");
        _detectionInProgress = NO;
        self.isDetecting = NO;
        if (completion) {
            dispatch_async(dispatch_get_main_queue(), ^{
                completion(nil, [NSError errorWithDomain:@"MioFriendDetect"
                                                    code:-3
                                                userInfo:@{NSLocalizedDescriptionKey: @"未找到好友列表"}]);
            });
        }
        return;
    }

    // ② 过滤已检测过的好友（断点续传）
    NSMutableArray<NSDictionary *> *toCheck = [allFriends mutableCopy];
    NSSet<NSString *> *checkedSet = resumeFrom ? resumeFrom.checkedWxIDs : [NSSet set];
    if (checkedSet.count > 0) {
        NSMutableArray *filtered = [NSMutableArray array];
        for (NSDictionary *friend in toCheck) {
            NSString *wxID = friend[@"wxID"];
            if (![checkedSet containsObject:wxID]) {
                [filtered addObject:friend];
            }
        }
        toCheck = filtered;
        WPLog(@"FriendDetect", @"[Execute] Resumed: %lu already checked, %lu remaining",
              (unsigned long)checkedSet.count, (unsigned long)toCheck.count);
    }

    NSInteger totalCount = toCheck.count;
    NSMutableArray<MioFriendDetectResult *> *deletedFriends = [NSMutableArray array];
    NSMutableArray<MioFriendDetectResult *> *invalidFriends = [NSMutableArray array];
    NSMutableSet<NSString *> *allCheckedWxIDs = [checkedSet mutableCopy];

    // ③ 逐个检测
    for (NSInteger i = 0; i < totalCount; i++) {
        if (self.shouldStop) {
            WPLog(@"FriendDetect", @"[Execute] Stopped by user at %ld/%ld", (long)i, (long)totalCount);
            break;
        }

        NSDictionary *friend = toCheck[i];
        NSString *wxID = friend[@"wxID"];
        NSString *nickname = friend[@"nickname"];
        NSString *remark = friend[@"remark"];

        @autoreleasepool {
            WPLog(@"FriendDetect", @"[Execute] Checking %ld/%ld: %@(%@)", (long)(i+1), (long)totalCount, nickname, wxID);

            // 更新进度
            if (progress) {
                dispatch_async(dispatch_get_main_queue(), ^{
                    progress(wxID, nickname ?: wxID, i + 1, totalCount);
                });
            }

            // ③-a: 创建转账预下单请求
            id request = createTransferRequest(wxID);
            if (!request) {
                WPLog(@"FriendDetect", @"[Execute] Failed to create request for %@", wxID);
                [allCheckedWxIDs addObject:wxID];
                continue;
            }

            // ③-b: 设置全局检测状态
            _currentCheckingWxID = wxID;
            _currentCheckResponse = nil;
            _detectionSemaphore = dispatch_semaphore_create(0);

            // ③-c: 发送请求
            BOOL sent = sendTransferPrepayRequest(request);
            if (!sent) {
                WPLog(@"FriendDetect", @"[Execute] Failed to send request for %@", wxID);
                _detectionSemaphore = NULL;
                _currentCheckingWxID = nil;
                [allCheckedWxIDs addObject:wxID];
                continue;
            }

            // ③-d: 等待 Hook 回调（超时 10 秒）
            dispatch_time_t timeout = dispatch_time(DISPATCH_TIME_NOW, 10 * NSEC_PER_SEC);
            long waitResult = dispatch_semaphore_wait(_detectionSemaphore, timeout);

            // ③-e: 解析结果
            NSDictionary *response = nil;
            @synchronized (_currentCheckResponse ?: [NSNull null]) {
                response = _currentCheckResponse;
            }

            MioFriendDetectResult *result = nil;
            if (waitResult == 0 && response) {
                MioFriendStatus status = determineStatusFromResponse(response);
                NSInteger retcode = [response[@"retcode"] integerValue];
                NSString *retmsg = response[@"retmsg"] ?: @"";

                result = [MioFriendDetectResult resultWithWxID:wxID
                                                      nickname:nickname
                                                        remark:remark
                                                        status:status
                                                       retcode:retcode
                                                        retmsg:retmsg];

                if (status == MioFriendStatusDeleted) {
                    [deletedFriends addObject:result];
                } else if (status == MioFriendStatusInvalid) {
                    [invalidFriends addObject:result];
                }
            } else {
                // 超时或没有响应 → 标记为异常
                WPLog(@"FriendDetect", @"[Execute] Timeout/no response for %@", wxID);
                result = [MioFriendDetectResult resultWithWxID:wxID
                                                      nickname:nickname
                                                        remark:remark
                                                        status:MioFriendStatusInvalid
                                                       retcode:-1
                                                        retmsg:@"超时/无响应"];
                [invalidFriends addObject:result];
            }

            [allCheckedWxIDs addObject:wxID];
            WPLog(@"FriendDetect", @"[Execute] Result for %@: %@", wxID, result);

            // 清理
            _detectionSemaphore = NULL;
            _currentCheckingWxID = nil;
            _currentCheckResponse = nil;
        }

        // 间隔 100ms，避免请求过快
        [NSThread sleepForTimeInterval:0.1];
    }

    // ④ 组装结果
    MioFriendDetectSummary *summary = [[MioFriendDetectSummary alloc] init];
    summary.timestamp = [[NSDate date] timeIntervalSince1970];
    summary.totalCount = allFriends.count;
    summary.deletedFriends = [deletedFriends copy];
    summary.invalidFriends = [invalidFriends copy];
    summary.checkedWxIDs = [allCheckedWxIDs copy];

    // ⑤ 持久化
    [MioFriendDetector saveSummary:summary];

    // ⑥ 回调
    _detectionInProgress = NO;
    self.isDetecting = NO;

    WPLog(@"FriendDetect", @"========================================");
    WPLog(@"FriendDetect", @" Detection complete: %lu deleted, %lu invalid",
          (unsigned long)deletedFriends.count, (unsigned long)invalidFriends.count);
    WPLog(@"FriendDetect", @"========================================");

    if (completion) {
        dispatch_async(dispatch_get_main_queue(), ^{
            completion(summary, nil);
        });
    }
}

#pragma mark - 读取/保存/清除结果

+ (void)saveSummary:(MioFriendDetectSummary *)summary {
    if (!summary) return;
    NSData *data = nil;
    @try {
        if (@available(iOS 11.0, *)) {
            data = [NSKeyedArchiver archivedDataWithRootObject:summary
                                         requiringSecureCoding:YES
                                                         error:nil];
        } else {
            data = [NSKeyedArchiver archivedDataWithRootObject:summary];
        }
    } @catch (NSException *e) {
        WPLog(@"FriendDetect", @"[Save] Archive failed: %@", e);
        return;
    }
    if (data) {
        [[NSUserDefaults standardUserDefaults] setObject:data forKey:kSummaryKey];
        [[NSUserDefaults standardUserDefaults] synchronize];
        WPLog(@"FriendDetect", @"[Save] Saved: %lu deleted, %lu invalid",
              (unsigned long)summary.deletedFriends.count,
              (unsigned long)summary.invalidFriends.count);
    }
}

+ (MioFriendDetectSummary *)loadSavedSummary {
    NSData *data = [[NSUserDefaults standardUserDefaults] objectForKey:kSummaryKey];
    if (!data) return nil;

    @try {
        if (@available(iOS 11.0, *)) {
            return [NSKeyedUnarchiver unarchivedObjectOfClass:[MioFriendDetectSummary class]
                                                     fromData:data
                                                        error:nil];
        } else {
            return [NSKeyedUnarchiver unarchiveObjectWithData:data];
        }
    } @catch (NSException *e) {
        WPLog(@"FriendDetect", @"[Load] Unarchive failed: %@", e);
        return nil;
    }
}

+ (void)clearSavedSummary {
    [[NSUserDefaults standardUserDefaults] removeObjectForKey:kSummaryKey];
    [[NSUserDefaults standardUserDefaults] synchronize];
    WPLog(@"FriendDetect", @"[Clear] Saved summary cleared");
}

    NSArray *friends = getAllFriends();

// ============================================================
// MARK: - 安装 WCPayLogicMgr Hook
// ============================================================

static IMP _orig_OnGetTransferPrepayRespone = NULL;
static IMP _orig_insideCallback = NULL;

// 全局检测状态供 Hook 回调使用
static dispatch_semaphore_t g_detectionSemaphore = NULL;
static NSString *g_currentWxID = nil;
static NSDictionary *g_currentResponse = nil;
static volatile BOOL g_detectionActive = NO;

/*
 * 以下 Hook 回调函数必须在 FriendDetectionHook.m 中定义为全局函数
 * 原因是 MSHookMessageEx 需要静态函数地址
 */

static void hook_OnGetTransferPrepayRespone(id self, SEL _cmd, id response) {
    if (_orig_OnGetTransferPrepayRespone) {
        ((void (*)(id, SEL, id))_orig_OnGetTransferPrepayRespone)(self, _cmd, response);
    }
    if (!g_detectionActive || !response) return;

    NSMutableDictionary *result = [NSMutableDictionary dictionary];
    if ([response isKindOfClass:[NSDictionary class]]) {
        NSDictionary *d = (NSDictionary *)response;
        result[@"retcode"] = d[@"retcode"];
        result[@"retmsg"] = d[@"retmsg"];
        result[@"wx_error_msg"] = d[@"wx_error_msg"];
    } else {
        if ([response respondsToSelector:sel_registerName("retcode")])
            result[@"retcode"] = @(((NSInteger (*)(id, SEL))objc_msgSend)(response, sel_registerName("retcode")));
        if ([response respondsToSelector:sel_registerName("retmsg")])
            result[@"retmsg"] = ((id (*)(id, SEL))objc_msgSend)(response, sel_registerName("retmsg"));
        if ([response respondsToSelector:sel_registerName("wx_error_msg")])
            result[@"wx_error_msg"] = ((id (*)(id, SEL))objc_msgSend)(response, sel_registerName("wx_error_msg"));
    }

    @synchronized (result) {
        g_currentResponse = result;
    }
    if (g_detectionSemaphore) {
        dispatch_semaphore_signal(g_detectionSemaphore);
    }
}

static void hook_insideCallback(id self, SEL _cmd, id response, id request) {
    if (_orig_insideCallback) {
        ((void (*)(id, SEL, id, id))_orig_insideCallback)(self, _cmd, response, request);
    }
    if (!g_detectionActive || g_currentResponse) return;

    if ([response isKindOfClass:[NSDictionary class]]) {
        g_currentResponse = response;
    }
    if (g_detectionSemaphore) {
        dispatch_semaphore_signal(g_detectionSemaphore);
    }
}

    // ========================================

    if (payCls) {
        SEL sel1 = sel_registerName("OnGetTransferPrepayRespone:");
        Method m1 = class_getInstanceMethod(payCls, sel1);
        if (m1) {
            MSHookMessageEx(payCls, sel1, (IMP)hook_OnGetTransferPrepayRespone, &_orig_OnGetTransferPrepayRespone);
