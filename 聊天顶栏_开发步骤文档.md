```objc
// 123456.c L12926: 使用自定义 CSUserInfoPopoverController（123456.c 内部类，50+ 方法）
void CSChatAvatarTitleView::presentUserInfoPopoverWithContact_sourceView_
               (ID self, SEL sel, id contact, id sourceView) {
    // 1. 创建 CSUserInfoPopoverController（自定义 popover 控制器）
    CSUserInfoPopoverController *popover = [[CSUserInfoPopoverController alloc] init];
    
    // 2. 传入 contact（通过 setupWithContact: setter）
    [popover setupWithContact:contact];
    
    // 3. 获取当前 VC
    UIViewController *parentVC = [self findViewController];
    if (parentVC) {
        // 4. 配置为 popover 样式
        popover.modalPresentationStyle = 7;  // UIModalPresentationPopover?
        popover.preferredContentSize = CGSizeMake(280, 400);
        
        // 5. 设置 sourceView/sourceRect
        UIPopoverPresentationController *ppc = popover.popoverPresentationController;
        ppc.sourceView = sourceView ?: self;
        ppc.sourceRect = sourceView ? sourceView.bounds : self.bounds;
        ppc.permittedArrowDirections = 15;  // UIPopoverArrowDirectionAny
```

### 关键区别总结

| 对比项 | 项目（错误） | 123456.c（正确） |
|--------|-------------|-----------------|
| 控制器 | WCUserInfoViewController / ContactInfoViewController | CSUserInfoPopoverController（自定义内部类） |
| 参数类型 | `NSString *wxid` | `id contact`（CContact 对象） |
| 展示方式 | `pushViewController:` | `presentViewController:` (popover) |
| Fallback | `initWithContact:` → crash | 无 fallback（不需要） |

### 修复方案

由于 CSUserInfoPopoverController 是微信优化的内部类（50+ 个方法，实现了完整的头像、昵称、备注、性别、地区、签名、群信息、公众号信息等展示），项目中没有此代码，无法直接调用。

**推荐修复**：替换为微信自带的方案，确保不 crash：

```objc
- (void)presentUserInfoPopoverWithContact:(id)contact sourceView:(UIView *)sourceView {
    // 注意：参数改为 id contact（CContact 对象），不是 NSString *wxid
    if (!contact) return;

    // 方案 A：尝试 WCUserInfoViewController（微信内部类，需要 CContact 对象）
    Class infoVCClass = objc_getClass("WCUserInfoViewController");
    if (infoVCClass) {
        NSString *wxid = ((id (*)(id, SEL))objc_msgSend)(contact, NSSelectorFromString(@"m_nsUsrName"));
        id infoVC = ((id (*)(Class, SEL))objc_msgSend)(infoVCClass, NSSelectorFromString(@"alloc"));
        infoVC = ((id (*)(id, SEL, id, id))objc_msgSend)(infoVC,
            NSSelectorFromString(@"initWithUsrName:contact:"), wxid, contact);
        if (infoVC) {
            UIViewController *parentVC = [self findViewController];
            if (parentVC && parentVC.navigationController) {
                [parentVC.navigationController pushViewController:infoVC animated:YES];
                [infoVC release];
                return;
            }
            [infoVC release];
        }
    }
    
    // ★ 删除 initWithContact: 的 fallback（ContactInfoViewController 没有此方法）
    // 如果 WCUserInfoViewController 也不可用，什么都不做（不 crash）
}

// 同时修改 Tap 手势处理，传 contact 对象而非 wxid 字符串：
- (void)onLeftAvatarTapped:(UITapGestureRecognizer *)gesture {
    // ... haptic feedback ...
    id contact = ((id (*)(id, SEL))objc_msgSend)(self.chatController, NSSelectorFromString(@"GetContact"));
    [self presentUserInfoPopoverWithContact:contact sourceView:self.leftAvatarView];
}

- (void)onRightAvatarTapped:(UITapGestureRecognizer *)gesture {
    // ... haptic feedback ...
    // 从 CContactMgr 获取 self contact（对齐 123456.c）
    Class serviceCenter = objc_getClass("MMServiceCenter");
    id center = ((id (*)(Class, SEL))objc_msgSend)(serviceCenter, NSSelectorFromString(@"defaultCenter"));
    id contactMgr = ((id (*)(id, SEL, Class))objc_msgSend)(center, NSSelectorFromString(@"getService:"),
        objc_getClass("CContactMgr"));
    id selfContact = ((id (*)(id, SEL))objc_msgSend)(contactMgr, @selector(getSelfContact));
    [self presentUserInfoPopoverWithContact:selfContact sourceView:self.rightAvatarView];
}
```

## 严重 Bug #2：loadWeChatAvatarForWxid: 用错方法 → 头像不加载

> 涉及文件：[CSChatAvatarTitleView.m](file:///www/wwwroot/ios/WeChatPlugin/Modules/ChatTopBar/CSChatAvatarTitleView.m) L434-L451

**这是头像不加载的根因！**

### 项目当前代码（错误）

```objc
- (UIImage *)loadWeChatAvatarForWxid:(NSString *)wxid {
    // 1. 获取 CContactMgr
    id contactMgr = [[MMServiceCenter defaultCenter] getService:[CContactMgr class]];
    // 2. 查找 contact → m_avatarImage
    id contact = [contactMgr getContactByName:wxid];
    if ([contact respondsToSelector:@selector(m_avatarImage)]) {
        return [contact m_avatarImage];  // ★ 绝大多数情况下 = nil!
    }
    return nil;
}
```

**问题分析**：`m_avatarImage` 是 `CContact` 的内存缓存属性，只有在微信某处加载过该联系人头像时才有值。聊天页面通常不会主动触发热加载联系人头像到这个属性上，因此对大多数联系人返回 nil。

### 123456.c 的正确实现（L12180-12222）

```objc
// 123456.c 使用 MMHeadImageMgr（微信的头像管理器）
ID CSChatAvatarTitleView::loadWeChatAvatarForWxid_(ID self, SEL sel, id wxid) {
    id center = [[MMServiceCenter defaultCenter] retain];
    if (!center) return nil;

    // ★ 使用 MMHeadImageMgr 而非 CContactMgr
    id headImageMgr = [center getService:[MMHeadImageMgr class]];
    if (!headImageMgr) { [center release]; return nil; }

    // ★ 调用 getHeadImage:withCategory: 实际获取头像
    if (![headImageMgr respondsToSelector:@selector(getHeadImage:withCategory:)]) {
        [headImageMgr release];
        [center release];
        return nil;
    }

    UIImage *image = [headImageMgr getHeadImage:wxid 
                                   withCategory:[NSNumber numberWithInt:0]];
    [headImageMgr release];
    [center release];
    return image;
}
```

**关键：`MMHeadImageMgr.getHeadImage:withCategory:` 会主动从磁盘缓存/网络获取头像图片，而不是依赖 CContact 对象的内存缓存。**

### 修复

```objc
- (UIImage *)loadWeChatAvatarForWxid:(NSString *)wxid {
    Class serviceCenter = objc_getClass("MMServiceCenter");
    if (!serviceCenter) return nil;

    id center = ((id (*)(Class, SEL))objc_msgSend)(serviceCenter, NSSelectorFromString(@"defaultCenter"));
    
    // ★ 改为使用 MMHeadImageMgr
    id headImageMgr = ((id (*)(id, SEL, Class))objc_msgSend)(center, NSSelectorFromString(@"getService:"), 
        objc_getClass("MMHeadImageMgr"));
    if (!headImageMgr) return nil;

    if (!((BOOL (*)(id, SEL, SEL))objc_msgSend)(headImageMgr, @selector(respondsToSelector:), 
        NSSelectorFromString(@"getHeadImage:withCategory:"))) {
        return nil;
    }

    // getHeadImage:withCategory: 第二个参数需要 NSNumber
    UIImage *image = ((UIImage *(*)(id, SEL, id, id))objc_msgSend)(headImageMgr,
        NSSelectorFromString(@"getHeadImage:withCategory:"), wxid, @0);
    return image;
}
```

> 涉及文件：[CSChatAvatarTitleView.m](file:///www/wwwroot/ios/WeChatPlugin/Modules/ChatTopBar/CSChatAvatarTitleView.m) L411-L432

### 项目当前代码


### 123456.c 的正确实现（L12143）

```objc

```

`createAnimatedImageFromGIFData`（L10819）是微信优化内部的函数，使用 CGImageSource 逐帧读取 GIF，创建 `animatedImageWithImages:duration:`。

### 修复

```objc
// 新增函数
- (UIImage *)createAnimatedImageFromGIFData:(NSData *)data {
    if (!data) return nil;
    CGImageSourceRef source = CGImageSourceCreateWithData((__bridge CFDataRef)data, NULL);
    if (!source) return nil;
    
    size_t count = CGImageSourceGetCount(source);
    if (count < 2) {
        CFRelease(source);
        return [UIImage imageWithData:data];
    }
    
    NSMutableArray *images = [NSMutableArray array];
    NSTimeInterval totalDuration = 0;
    
    for (size_t i = 0; i < count; i++) {
        CGImageRef cgImage = CGImageSourceCreateImageAtIndex(source, i, NULL);
        if (!cgImage) continue;
        
        NSDictionary *props = (__bridge_transfer NSDictionary *)CGImageSourceCopyPropertiesAtIndex(source, i, NULL);
        NSDictionary *gifProps = props[(__bridge NSString *)kCGImagePropertyGIFDictionary];
        NSNumber *delay = gifProps[(__bridge NSString *)kCGImagePropertyGIFUnclampedDelayTime];
        if (!delay) delay = gifProps[(__bridge NSString *)kCGImagePropertyGIFDelayTime];
        NSTimeInterval dt = delay ? [delay doubleValue] : 0.1;
        totalDuration += dt;
        
        [images addObject:[UIImage imageWithCGImage:cgImage]];
        CGImageRelease(cgImage);
    }
    
    CFRelease(source);
    if (images.count == 0) return nil;
    return [UIImage animatedImageWithImages:images duration:totalDuration];
}

// 然后在 loadCustomAvatarForWxid: 中：
NSData *data = [NSData dataWithContentsOfFile:gifPath];
return [self createAnimatedImageFromGIFData:data];
```
