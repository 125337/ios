### 5.2 ChatTopBarHook.m — 完整实现

```objc
#import "ChatTopBarHook.h"
#import "CSChatAvatarTitleView.h"
#import "../../Config/PluginConfig.h"
#import "../../Core/LogManager.h"
#import <substrate.h>
#import <objc/runtime.h>
#import <objc/message.h>

// Original IMPs
static IMP _orig_BaseMsgContentVC_viewDidLoad   = NULL;
static IMP _orig_BaseMsgContentVC_viewWillAppear = NULL;

// associatedObject key
static const void *kOriginalTitleViewKey = &kOriginalTitleViewKey;

// ============================================================
// 黑名单匹配 (对齐 FUN_00013157)
// ============================================================
static BOOL isContactInBlacklist(NSString *contactID) {
    if (!contactID.length) return NO;
    NSString *blacklist = [[NSUserDefaults standardUserDefaults]
        stringForKey:[kPluginPrefix stringByAppendingString:@"ChatAvatarBlacklist"]];
    if (!blacklist.length) return NO;

    NSArray *lines = [blacklist componentsSeparatedByString:@"\n"];
    BOOL isGroup = [contactID containsString:@"@chatroom"];

    for (NSString *line in lines) {
        NSString *t = [line stringByTrimmingCharactersInSet:
                        [NSCharacterSet whitespaceCharacterSet]];
        if (!t.length) continue;
        if (isGroup && [t isEqualToString:@"@"]) return YES;
        if ([contactID hasPrefix:@"gh_"] && [t hasPrefix:@"gh_"]) return YES;
        if ([t isEqualToString:contactID]) return YES;
        if (isGroup && [contactID containsString:t]) return YES;
    }
    return NO;
}

// ============================================================
// Hook: viewDidLoad (对齐 FUN_00014c74 L13285)
// ============================================================
static void hook_viewDidLoad(id self, SEL _cmd) {
    // 调用原始方法
    ((void (*)(id, SEL))_orig_BaseMsgContentVC_viewDidLoad)(self, _cmd);

    PluginConfig *config = [PluginConfig shared];
    if (!config.showChatAvatar) return;

    // 检查是否已保存过原始 titleView
    uintptr_t saved = (uintptr_t)objc_getAssociatedObject(self, kOriginalTitleViewKey);
    if (saved == 0) {
        // 首次进入，保存原始 titleView
        id originalTitleView = [self navigationItem].titleView;
        if (originalTitleView) {
            objc_setAssociatedObject(self, kOriginalTitleViewKey,
                                     (__bridge id)(void *)originalTitleView,
                                     OBJC_ASSOCIATION_ASSIGN);
        }
    }

    // 获取联系人
    id contact = ((id (*)(id, SEL))objc_msgSend)(self, NSSelectorFromString(@"GetContact"));
    if (!contact) return;

    NSString *username = ((id (*)(id, SEL))objc_msgSend)(contact, NSSelectorFromString(@"m_nsUsrName"));
    if (isContactInBlacklist(username)) return;

    // 创建 CSChatAvatarTitleView
    CGFloat width = config.chatTitleViewWidth > 0 ? config.chatTitleViewWidth : 210.0;
    CSChatAvatarTitleView *view = [[CSChatAvatarTitleView alloc]
                                    initWithFrame:CGRectMake(0, 0, width, 45)];
    [view setChatController:(__bridge id)(__bridge_retained void *)self]; // weak, store it properly

    [[self navigationItem].titleView removeFromSuperview];
    [[self navigationItem] setTitleView:view];
    [view layoutSubviews];
}

// ============================================================
// Hook: viewWillAppear: (对齐 FUN_00014ed0 L13357)
// ============================================================
static void hook_viewWillAppear(id self, SEL _cmd, BOOL animated) {
    ((void (*)(id, SEL, BOOL))_orig_BaseMsgContentVC_viewWillAppear)(self, _cmd, animated);

    PluginConfig *config = [PluginConfig shared];
    id currentTitle = [[self navigationItem] titleView];

    if (!config.showChatAvatar) {
        // 开关关闭 → 恢复原始 titleView
        if ([currentTitle isKindOfClass:[CSChatAvatarTitleView class]]) {
            uintptr_t saved = (uintptr_t)objc_getAssociatedObject(self, kOriginalTitleViewKey);
            if (saved != 0) {
                UIView *originalTitle = (__bridge UIView *)(void *)saved;
                [[self navigationItem] setTitleView:originalTitle];
            }
        }
        return;
    }

    // 开关打开 → 黑名单检查
    id contact = ((id (*)(id, SEL))objc_msgSend)(self, NSSelectorFromString(@"GetContact"));
    if (!contact) return;
    NSString *username = ((id (*)(id, SEL))objc_msgSend)(contact, NSSelectorFromString(@"m_nsUsrName"));
    if (isContactInBlacklist(username)) {
        // 在黑名单中 → 恢复原始
        uintptr_t saved = (uintptr_t)objc_getAssociatedObject(self, kOriginalTitleViewKey);
        if (saved != 0 && [currentTitle isKindOfClass:[CSChatAvatarTitleView class]]) {
            [[self navigationItem] setTitleView:(__bridge UIView *)(void *)saved];
        }
        return;
    }

    // 确保 titleView 是 CSChatAvatarTitleView
    if (![currentTitle isKindOfClass:[CSChatAvatarTitleView class]]) {
        CGFloat width = config.chatTitleViewWidth > 0 ? config.chatTitleViewWidth : 210.0;
        CSChatAvatarTitleView *view = [[CSChatAvatarTitleView alloc]
                                        initWithFrame:CGRectMake(0, 0, width, 45)];
        [view setChatController:(__bridge id)(__bridge_retained void *)self];
        [[self navigationItem] setTitleView:view];
        currentTitle = view;
    }

    // 刷新
    [(CSChatAvatarTitleView *)currentTitle updateAvatars];
    [currentTitle setNeedsLayout];
    [currentTitle layoutIfNeeded];
    [currentTitle setNeedsDisplay];
}

// ============================================================
// Install
// ============================================================
@implementation ChatTopBarHook

+ (void)install {
    WPLog(@"ChatTopBar", @"ChatTopBarHook install");

    Class cls = objc_getClass("BaseMsgContentViewController");
    if (cls) {
        MSHookMessageEx(cls, @selector(viewDidLoad),
                        (IMP)hook_viewDidLoad,
                        &_orig_BaseMsgContentVC_viewDidLoad);
        MSHookMessageEx(cls, @selector(viewWillAppear:),
                        (IMP)hook_viewWillAppear,
                        &_orig_BaseMsgContentVC_viewWillAppear);
        WPLog(@"ChatTopBar", @"[Hook] ✓ BaseMsgContentViewController");
    }
}

@end
```

### 5.3 与 HideAvatarHook 的冲突处理

> **重要**：`HideAvatarHook` 已经 Hook 了 `BaseMsgContentViewController.viewDidLoad` 和 `viewWillAppear:`（见 [HideAvatarHook.m L199-L215](file:///www/wwwroot/ios/WeChatPlugin/Modules/HideAvatar/HideAvatarHook.m#L199-L215)）。两个模块 Hook 同一方法时 `MSHookMessageEx` 会**链式调用**（后安装的 Hook 调用前面保存的原 IMP），默认是兼容的。

**方案**：不需要修改 HideAvatarHook。只需确保 FeatureModuleRegistry 中 `ChatTopBarHook` 的 install 排在 `HideAvatarHook` 之后（或在 layout 模块中 install），这样两个 Hook 都能正常工作。

### 7.1 让 SettingChatTopBarController 实现 UIImagePickerControllerDelegate

在 `@implementation SettingChatTopBarController` 中追加：

```objc
#pragma mark - UIImagePickerControllerDelegate

- (void)imagePickerController:(UIImagePickerController *)picker
didFinishPickingMediaWithInfo:(NSDictionary<UIImagePickerControllerInfoKey, id> *)info {
    UIImage *image = info[UIImagePickerControllerOriginalImage];
    if (!image) {
        [picker dismissViewControllerAnimated:YES completion:nil];
        return;
    }

    PluginConfig *config = [PluginConfig shared];
    NSString *saveKey = nil;

    // 根据 picker 的 mediaTypes 区分静态图还是 GIF
    // 简单方案：通过 tag 区分
    if (picker.view.tag == 100) {
        // 静态图片
        saveKey = @"ChatSeparatorIcon";
        NSData *pngData = UIImagePNGRepresentation(image);
        [[NSUserDefaults standardUserDefaults] setObject:pngData
                                                  forKey:[kPluginPrefix stringByAppendingString:saveKey]];
    } else if (picker.view.tag == 200) {
        // GIF (需要从 info 中获取 GIF 数据)
        saveKey = @"ChatSeparatorGIF";
        NSData *gifData = [NSData dataWithContentsOfURL:info[UIImagePickerControllerImageURL]];
        if (gifData) {
            [[NSUserDefaults standardUserDefaults] setObject:gifData
                                                      forKey:[kPluginPrefix stringByAppendingString:saveKey]];
        }
    }

    [[NSUserDefaults standardUserDefaults] synchronize];
    [picker dismissViewControllerAnimated:YES completion:^{
        [self buildUI];
    }];
}

- (void)imagePickerControllerDidCancel:(UIImagePickerController *)picker {
    [picker dismissViewControllerAnimated:YES completion:nil];
}
```

## Step 8：黑名单编辑器数据贯通

> 涉及文件：[ChatTopBarBlacklistEditorVC.m](file:///www/wwwroot/ios/WeChatPlugin/Settings/Controllers/ChatTopBarBlacklistEditorVC.m)

当前 `ChatTopBarBlacklistEditorVC` 的 `saveAction` 没有实际保存数据。需要将 editorView 的内容写入 NSUserDefaults。

### 8.1 viewDidLoad 追加 — 加载已有黑名单

```objc
// 在 [self registerKeyboardNotifications] 之前插入：
NSString *saved = [[NSUserDefaults standardUserDefaults]
    stringForKey:[kPluginPrefix stringByAppendingString:@"ChatAvatarBlacklist"]];
if (saved.length > 0) {
    self.editorView.text = saved;
}
```

### 8.2 修改 saveAction

```objc
- (void)saveAction {
    NSString *text = self.editorView.text ?: @"";
    [[NSUserDefaults standardUserDefaults] setObject:text
                                              forKey:[kPluginPrefix stringByAppendingString:@"ChatAvatarBlacklist"]];
    [[NSUserDefaults standardUserDefaults] synchronize];
    [self dismissViewControllerAnimated:YES completion:nil];
}
```

> 需要 import `<objc/runtime.h>` 或 Constants.h 以引入 `kPluginPrefix`。
